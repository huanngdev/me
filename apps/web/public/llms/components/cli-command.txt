## packages/core/src/components/cli-command/package-managers.ts
```ts
export const PACKAGE_MANAGERS = ["npm", "pnpm", "yarn", "bun"] as const;

export type PackageManager = (typeof PACKAGE_MANAGERS)[number];

export function isPackageManager(value: string | null): value is PackageManager {
  return value !== null && (PACKAGE_MANAGERS as readonly string[]).includes(value);
}

export type CliCommandSource =
  | { install: string; dev?: boolean }
  | { execute: string }
  | { run: string }
  | { commands: Record<PackageManager, string> };

const INSTALLERS: Record<PackageManager, string> = {
  npm: "npm install",
  pnpm: "pnpm add",
  yarn: "yarn add",
  bun: "bun add",
};

const EXECUTORS: Record<PackageManager, string> = {
  npm: "npx",
  pnpm: "pnpm dlx",
  yarn: "yarn dlx",
  bun: "bunx --bun",
};

const RUNNERS: Record<PackageManager, string> = {
  npm: "npm run",
  pnpm: "pnpm",
  yarn: "yarn",
  bun: "bun run",
};

function normalize(value: string): string {
  return value
    .split("\n")
    .map((line) => line.trim().replace(/[ \t]+/g, " "))
    .join("\n")
    .trim();
}

function withVerb(verb: string, body: string): string {
  return body === "" ? verb : `${verb} ${body}`;
}

function perManager(build: (manager: PackageManager) => string): Record<PackageManager, string> {
  return {
    npm: build("npm"),
    pnpm: build("pnpm"),
    yarn: build("yarn"),
    bun: build("bun"),
  };
}

export function resolveCommands(source: CliCommandSource): Record<PackageManager, string> {
  if ("commands" in source) {
    return { ...source.commands };
  }
  if ("install" in source) {
    const packages = normalize(source.install);
    const flag = source.dev === true ? " -D" : "";
    return perManager((manager) => withVerb(`${INSTALLERS[manager]}${flag}`, packages));
  }
  if ("execute" in source) {
    const command = normalize(source.execute);
    return perManager((manager) => withVerb(EXECUTORS[manager], command));
  }
  const script = normalize(source.run);
  return perManager((manager) => withVerb(RUNNERS[manager], script));
}
```

## packages/core/src/components/cli-command/shell-tokens.ts
```ts
export type ShellTokenType = "program" | "flag" | "string" | "operator" | "text" | "space";

export type ShellToken = {
  type: ShellTokenType;
  value: string;
};

const OPERATORS = new Set([";", "|"]);

function startsToken(line: string, index: number): boolean {
  const char = line[index];
  if (char === " " || char === "\t" || char === "'" || char === '"') {
    return true;
  }
  if (char !== undefined && OPERATORS.has(char)) {
    return true;
  }
  return char === "&" && line[index + 1] === "&";
}

/**
 * Splits one command line into colored spans. Concatenating every `value` gives the input back
 * byte for byte, so the rendered text never differs from the copied text.
 */
export function tokenizeCommand(line: string): ShellToken[] {
  const tokens: ShellToken[] = [];
  let index = 0;
  let nextIsProgram = true;

  while (index < line.length) {
    const char = line[index] as string;

    if (char === " " || char === "\t") {
      let value = "";
      while (index < line.length && (line[index] === " " || line[index] === "\t")) {
        value += line[index];
        index += 1;
      }
      tokens.push({ type: "space", value });
      continue;
    }

    if (char === "'" || char === '"') {
      let value = char;
      index += 1;
      while (index < line.length && line[index] !== char) {
        value += line[index];
        index += 1;
      }
      if (index < line.length) {
        value += line[index];
        index += 1;
      }
      tokens.push({ type: "string", value });
      nextIsProgram = false;
      continue;
    }

    if (OPERATORS.has(char) || (char === "&" && line[index + 1] === "&")) {
      let value = char;
      index += 1;
      if (index < line.length && line[index] === char) {
        value += line[index];
        index += 1;
      }
      tokens.push({ type: "operator", value });
      nextIsProgram = true;
      continue;
    }

    let value = "";
    while (index < line.length && !startsToken(line, index)) {
      value += line[index];
      index += 1;
    }
    const type: ShellTokenType = nextIsProgram
      ? "program"
      : value.startsWith("-")
        ? "flag"
        : "text";
    tokens.push({ type, value });
    nextIsProgram = false;
  }

  return tokens;
}
```

## packages/core/src/components/cli-command/use-package-manager.ts
```ts
"use client";

import { useSyncExternalStore } from "react";

import { isPackageManager, type PackageManager } from "./package-managers";

const STORAGE_KEY = "package-manager";
const CHANGE_EVENT = "package-manager-change";

function readStored(): PackageManager | null {
  try {
    const value = window.localStorage.getItem(STORAGE_KEY);
    return isPackageManager(value) ? value : null;
  } catch {
    return null;
  }
}

function subscribe(onChange: () => void): () => void {
  window.addEventListener("storage", onChange);
  window.addEventListener(CHANGE_EVENT, onChange);
  return () => {
    window.removeEventListener("storage", onChange);
    window.removeEventListener(CHANGE_EVENT, onChange);
  };
}

/**
 * The selected package manager, shared by every `CliCommand` on the page and across browser tabs.
 * The server renders the default so the first paint matches the markup.
 */
export function usePackageManager(
  defaultManager: PackageManager,
): [PackageManager, (manager: PackageManager) => void] {
  const manager = useSyncExternalStore(
    subscribe,
    () => readStored() ?? defaultManager,
    () => defaultManager,
  );

  const setManager = (next: PackageManager): void => {
    try {
      window.localStorage.setItem(STORAGE_KEY, next);
    } catch {
      // Storage can be blocked; the in-page event still updates every command.
    }
    window.dispatchEvent(new Event(CHANGE_EVENT));
  };

  return [manager, setManager];
}
```

```tsx
"use client";

import { Terminal } from "lucide-react";

import { cn } from "../../lib/utils";
import { CopyButton } from "../copy-button";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "../tabs";
import { PACKAGE_MANAGER_ICONS } from "./package-manager-icons";
import {
  isPackageManager,
  PACKAGE_MANAGERS,
  resolveCommands,
  type CliCommandSource,
  type PackageManager,
} from "./package-managers";
import { tokenizeCommand, type ShellTokenType } from "./shell-tokens";
import { usePackageManager } from "./use-package-manager";

export type CliCommandProps = (CliCommandSource | { command: string }) & {
  defaultManager?: PackageManager;
  prompt?: string;
  className?: string;
};

const DEFAULT_PROMPT = "$";

const TOKEN_CLASS: Record<ShellTokenType, string> = {
  program: "font-medium text-foreground",
  flag: "text-primary",
  string: "text-muted-foreground",
  operator: "text-muted-foreground",
  text: "text-foreground/80",
  space: "",
};

const CARD_CLASS = "@container bg-background overflow-hidden rounded-xl border";

// The active tab's `::after` must sit inside the trigger so `overflow-x-auto` on the list cannot
// clip it. The primitive positions it at `bottom-[-5px]`; `-bottom-px` lands it on the header
// border (the trigger's transparent 1px border offsets the absolutely positioned box).
// The list repeats the primitive's `group-data-horizontal/tabs:` variant on height and width so
// tailwind-merge drops the primitive's `h-8`/`w-fit` for one trigger-filled row.
const TAB_LIST_CLASS =
  "no-scrollbar group-data-horizontal/tabs:h-full w-full min-w-0 flex-1 justify-start gap-1 overflow-x-auto p-0 [&_[data-slot=tabs-trigger][data-state=active]]:after:-bottom-px";
const TRIGGER_CLASS = "h-full flex-none gap-1 [&>svg]:hidden @min-[20rem]:[&>svg]:block";

function CommandBody({ command, prompt }: { command: string; prompt: string }) {
  return (
    <pre className="overflow-x-auto p-4 font-mono text-sm whitespace-pre">
      <code>
        {command.split("\n").map((line, lineIndex) => (
          <span key={lineIndex} className="block">
            <span aria-hidden className="text-muted-foreground select-none">
              {prompt}{" "}
            </span>
            {tokenizeCommand(line).map((token, tokenIndex) => (
              <span key={tokenIndex} className={TOKEN_CLASS[token.type]}>
                {token.value}
              </span>
            ))}
          </span>
        ))}
      </code>
    </pre>
  );
}

function PlainCommand({
  command,
  prompt,
  className,
}: {
  command: string;
  prompt: string;
  className?: string;
}) {
  return (
    <div className={cn(CARD_CLASS, className)}>
      <header className="flex h-10 shrink-0 items-center justify-between border-b pr-2 pl-3">
        <Terminal className="text-muted-foreground size-4" />
        <CopyButton text={command} size="icon-sm" aria-label="Copy command" className="shrink-0" />
      </header>
      <CommandBody command={command} prompt={prompt} />
    </div>
  );
}

function PackageManagerCommand({
  source,
  defaultManager,
  prompt,
  className,
}: {
  source: CliCommandSource;
  defaultManager: PackageManager;
  prompt: string;
  className?: string;
}) {
  const [manager, setManager] = usePackageManager(defaultManager);
  const commands = resolveCommands(source);

  return (
    <Tabs
      value={manager}
      onValueChange={(value) => {
        if (isPackageManager(value)) setManager(value);
      }}
      className={cn(CARD_CLASS, "gap-0", className)}
    >
      <header className="flex h-10 shrink-0 items-center justify-between gap-2 border-b pr-2 pl-1.5">
        <TabsList variant="line" className={TAB_LIST_CLASS}>
          {PACKAGE_MANAGERS.map((name) => {
            const Icon = PACKAGE_MANAGER_ICONS[name];
            return (
              <TabsTrigger key={name} value={name} className={TRIGGER_CLASS}>
                <Icon />
                {name}
              </TabsTrigger>
            );
          })}
        </TabsList>
        <CopyButton
          text={commands[manager]}
          size="icon-sm"
          aria-label="Copy command"
          className="shrink-0"
        />
      </header>
      {PACKAGE_MANAGERS.map((name) => (
        <TabsContent key={name} value={name}>
          <CommandBody command={commands[name]} prompt={prompt} />
        </TabsContent>
      ))}
    </Tabs>
  );
}

export function CliCommand({ prompt = DEFAULT_PROMPT, className, ...rest }: CliCommandProps) {
  if ("command" in rest) {
    return <PlainCommand command={rest.command} prompt={prompt} className={className} />;
  }
  return (
    <PackageManagerCommand
      source={rest}
      defaultManager={rest.defaultManager ?? "bun"}
      prompt={prompt}
      className={className}
    />
  );
}
```
