## packages/core/src/components/github-contributions-3d.utils.ts
```ts
import type { ContributionDay } from "./github-contributions-3d";

const DAY_IN_MS = 86_400_000;
const DAYS_PER_WEEK = 7;

export type ContributionCell = ContributionDay & {
  dayIndex: number;
  height: number;
  weekIndex: number;
};

export const createContributionCells = (data: ContributionDay[]): ContributionCell[] => {
  if (data.length === 0) return [];

  const sorted = [...data].sort((a, b) => a.date.localeCompare(b.date));
  const firstDate = sorted[0];
  if (!firstDate) return [];

  const firstTimestamp = Date.parse(`${firstDate.date}T00:00:00Z`);
  const firstDay = new Date(firstTimestamp).getUTCDay();

  return sorted.map((day) => {
    const date = new Date(`${day.date}T00:00:00Z`);
    const elapsedDays = Math.round((date.getTime() - firstTimestamp) / DAY_IN_MS);

    return {
      ...day,
      dayIndex: date.getUTCDay(),
      height: day.count === 0 ? 0.12 : Math.min(6.5, 0.22 + Math.log10(day.count / 2 + 1) * 2.8),
      weekIndex: Math.floor((firstDay + elapsedDays) / DAYS_PER_WEEK),
    };
  });
};
```

## packages/core/src/components/github-contributions-3d.tsx
```tsx
"use client";

import { Instances, Instance, OrbitControls, RoundedBox } from "@react-three/drei";
import { Canvas, useThree } from "@react-three/fiber";
import { useEffect, useState } from "react";

import { cn } from "../lib/utils";
import { createContributionCells, type ContributionCell } from "./github-contributions-3d.utils";

export type ContributionDay = {
  date: string;
  count: number;
  level: 0 | 1 | 2 | 3 | 4;
};

export type ContributionVariant =
  "green" | "season" | "night-view" | "night-green" | "night-rainbow" | "gitblock";

export type GitHubContributions3DProps = {
  data: ContributionDay[];
  name: string;
  variant?: ContributionVariant;
  className?: string;
};

const THEME = {
  green: {
    background: "#ffffff",
    base: "#e5e7eb",
    colors: ["#efefef", "#d8e887", "#8cc569", "#47a042", "#1d6a23"],
  },
  "night-view": {
    background: "#00000f",
    base: "#101b36",
    colors: [
      "rgb(25,60,130)",
      "rgb(25,90,210)",
      "rgb(25,120,220)",
      "rgb(25,150,230)",
      "rgb(25,165,240)",
    ],
  },
  "night-green": {
    background: "#00000f",
    base: "#171c19",
    colors: ["#444444", "#1b7d28", "#24a736", "#2dd143", "#57da69"],
  },
  gitblock: {
    background: "#ffffff",
    base: "#e5e7eb",
    colors: [
      "#f8f8f8",
      "hsl(125,52%,50%)",
      "hsl(242,100%,65%)",
      "hsl(48,100%,50%)",
      "hsl(350,100%,50%)",
    ],
  },
} as const;

const SEASON_COLORS = {
  spring: ["#efefef", "#ffe7ff", "#edaeda", "#e492ca", "#ba7aad"],
  summer: ["#efefef", "#d8e887", "#8cc569", "#47a042", "#1d6a23"],
  autumn: ["#efefef", "#ffed4a", "#ffc402", "#fe9400", "#fa6100"],
  winter: ["#efefef", "#999999", "#cccccc", "#dddddd", "#eeeeee"],
} as const;

const CELL_SIZE = 0.78;
const BASE_HEIGHT = 0.34;
const CAMERA_TARGET = [0, 1.1, 0] as const;
const DATE_FORMATTER = new Intl.DateTimeFormat("en-US", {
  day: "numeric",
  month: "short",
  timeZone: "UTC",
  year: "numeric",
});

const getSeasonColor = (cell: ContributionCell) => {
  const month = new Date(`${cell.date}T00:00:00Z`).getUTCMonth();
  if (month >= 2 && month <= 4) return SEASON_COLORS.spring[cell.level];
  if (month >= 5 && month <= 7) return SEASON_COLORS.summer[cell.level];
  if (month >= 8 && month <= 10) return SEASON_COLORS.autumn[cell.level];
  return SEASON_COLORS.winter[cell.level];
};

const getColor = (cell: ContributionCell, variant: ContributionVariant) => {
  if (variant === "season") return getSeasonColor(cell);

  if (variant === "night-rainbow") {
    const lightness = [20, 30, 35, 40, 50][cell.level];
    const hue = (((cell.weekIndex * -7) % 360) + 360) % 360;
    return `hsl(${hue}, 50%, ${lightness}%)`;
  }

  return THEME[variant].colors[cell.level];
};

const getThemeSurface = (variant: ContributionVariant) => {
  if (variant === "season") return THEME.green;
  if (variant === "night-rainbow") return THEME["night-view"];
  return THEME[variant];
};

const getLabelColor = (variant: ContributionVariant) => {
  if (variant === "season") return SEASON_COLORS.spring[4];
  if (variant === "night-rainbow") return "hsl(200, 50%, 70%)";
  return THEME[variant].colors[4];
};

function useWebGLSupport() {
  const [supported, setSupported] = useState<boolean | null>(null);

  useEffect(() => {
    const timeout = window.setTimeout(() => {
      const canvas = document.createElement("canvas");
      setSupported(Boolean(canvas.getContext("webgl2") || canvas.getContext("webgl")));
    });

    return () => window.clearTimeout(timeout);
  }, []);

  return supported;
}

function ResponsiveCamera() {
  const { camera, invalidate, size } = useThree();

  useEffect(() => {
    const [x, y, z] = size.width < 640 ? [22, 59, 89] : [14, 38, 58];
    camera.position.set(x, y, z);
    camera.lookAt(...CAMERA_TARGET);
    camera.updateProjectionMatrix();
    invalidate();
  }, [camera, invalidate, size.width]);

  return null;
}

function Scene({ cells, variant }: { cells: ContributionCell[]; variant: ContributionVariant }) {
  const weekCount = Math.max(...cells.map((cell) => cell.weekIndex), 0) + 1;
  const baseWidth = weekCount - 1 + CELL_SIZE;
  const baseDepth = 6 + CELL_SIZE;
  const surface = getThemeSurface(variant);

  return (
    <>
      <color attach="background" args={[surface.background]} />
      <ResponsiveCamera />
      <hemisphereLight args={["#ffffff", "#556070", 1.6]} />
      <directionalLight
        castShadow
        intensity={3.1}
        position={[-12, 18, 10]}
        shadow-bias={-0.0004}
        shadow-camera-bottom={-8}
        shadow-camera-far={70}
        shadow-camera-left={-30}
        shadow-camera-right={30}
        shadow-camera-top={12}
        shadow-mapSize-height={1024}
        shadow-mapSize-width={1024}
      />

      <group position={[-(weekCount - 1) / 2, 0, -3]}>
        <Instances castShadow limit={cells.length} range={cells.length} receiveShadow>
          <boxGeometry args={[CELL_SIZE, 1, CELL_SIZE]} />
          <meshStandardMaterial metalness={0.04} roughness={0.56} />
          {cells.map((cell) => (
            <Instance
              key={cell.date}
              color={getColor(cell, variant)}
              position={[cell.weekIndex, cell.height / 2, cell.dayIndex]}
              scale={[1, cell.height, 1]}
            />
          ))}
        </Instances>
      </group>

      <RoundedBox
        args={[baseWidth, BASE_HEIGHT, baseDepth]}
        position={[0, -BASE_HEIGHT / 2, 0]}
        radius={0.16}
        receiveShadow
        smoothness={4}
      >
        <meshStandardMaterial color={surface.base} metalness={0.02} roughness={0.78} />
      </RoundedBox>

      <OrbitControls
        enablePan={false}
        makeDefault
        maxDistance={130}
        maxPolarAngle={Math.PI / 2.15}
        minDistance={38}
        minPolarAngle={Math.PI / 5}
        target={CAMERA_TARGET}
      />
    </>
  );
}

export function GitHubContributions3D({
  data,
  name,
  variant = "green",
  className,
}: GitHubContributions3DProps) {
  const webGLSupported = useWebGLSupport();
  const cells = createContributionCells(data);
  const total = data.reduce((sum, day) => sum + day.count, 0);
  const firstDate = cells[0]?.date;
  const lastDate = cells.at(-1)?.date;
  const dateRange =
    firstDate && lastDate
      ? `${DATE_FORMATTER.format(new Date(`${firstDate}T00:00:00Z`))} – ${DATE_FORMATTER.format(
          new Date(`${lastDate}T00:00:00Z`),
        )}`
      : "";

  if (webGLSupported === false) {
    return (
      <div
        className={cn(
          "bg-muted flex min-h-80 items-center justify-center rounded-lg border p-6",
          className,
        )}
      >
        <p className="text-muted-foreground text-center text-sm">
          This browser cannot display the WebGL preview.
        </p>
      </div>
    );
  }

  if (cells.length === 0) {
    return (
      <div
        className={cn(
          "bg-muted flex min-h-80 items-center justify-center rounded-lg border p-6",
          className,
        )}
      >
        <p className="text-muted-foreground text-center text-sm">
          No contribution data to display.
        </p>
      </div>
    );
  }

  return (
    <div
      aria-label={`${name}, ${total.toLocaleString("en-US")} GitHub contributions from ${dateRange}, shown with the ${variant} theme`}
      className={cn(
        "bg-muted/30 relative h-80 w-full overflow-hidden rounded-lg border sm:h-112",
        className,
      )}
      role="img"
    >
      {webGLSupported && (
        <Canvas
          camera={{ far: 180, fov: 36, near: 0.1, position: [14, 38, 58] }}
          dpr={[1, 1.5]}
          frameloop="demand"
          gl={{ antialias: true, alpha: false }}
          shadows
        >
          <Scene cells={cells} variant={variant} />
        </Canvas>
      )}
      <div
        aria-hidden="true"
        className="pointer-events-none absolute bottom-3 left-3 font-mono leading-tight drop-shadow-sm"
        style={{ color: getLabelColor(variant) }}
      >
        <p className="text-xs font-medium">{name}</p>
        <p className="mt-1 text-[10px] opacity-80 sm:text-xs">{dateRange}</p>
      </div>
      <p className="text-muted-foreground bg-background/70 pointer-events-none absolute right-3 bottom-2 hidden rounded px-2 py-1 text-xs sm:block">
        Drag to rotate · Scroll to zoom
      </p>
    </div>
  );
}
```

```tsx
"use client";

import * as React from "react";
import { X } from "lucide-react";

import { Badge } from "../badge";
import { cn } from "../../lib/utils";

export type UseTagsInputOptions = {
  value?: string[];
  defaultValue?: string[];
  onChange?: (value: string[]) => void;
  maxTags?: number;
  minLength?: number;
  maxLength?: number;
  allowDuplicates?: boolean;
  trim?: boolean;
  validate?: (tag: string, tags: string[]) => true | string;
};

export type UseTagsInputReturn = {
  tags: string[];
  setTags: (value: string[]) => void;
  addTag: (tag: string) => true | string;
  removeTag: (index: number) => void;
  clear: () => void;
};

export function useTagsInput({
  value,
  defaultValue,
  onChange,
  maxTags,
  minLength = 1,
  maxLength,
  allowDuplicates = false,
  trim = true,
  validate,
}: UseTagsInputOptions = {}): UseTagsInputReturn {
  const isControlled = value !== undefined;
  const [internal, setInternal] = React.useState<string[]>(defaultValue ?? []);
  const tags = isControlled ? value : internal;

  const commit = React.useCallback(
    (next: string[]) => {
      if (!isControlled) setInternal(next);
      onChange?.(next);
    },
    [isControlled, onChange],
  );

  const setTags = React.useCallback((next: string[]) => commit(next), [commit]);

  const addTag = React.useCallback(
    (raw: string): true | string => {
      const tag = trim ? raw.trim() : raw;
      if (tag.length < minLength)
        return `Tag must be at least ${minLength} character${minLength === 1 ? "" : "s"}`;
      if (maxLength && tag.length > maxLength) return `Tag must be at most ${maxLength} characters`;
      if (!allowDuplicates && tags.includes(tag)) return "Tag already exists";
      if (maxTags && tags.length >= maxTags) return `Maximum ${maxTags} tags allowed`;
      if (validate) {
        const result = validate(tag, tags);
        if (result !== true) return result;
      }
      commit([...tags, tag]);
      return true;
    },
    [tags, trim, minLength, maxLength, allowDuplicates, maxTags, validate, commit],
  );

  const removeTag = React.useCallback(
    (index: number) => {
      if (index < 0 || index >= tags.length) return;
      const next = tags.slice();
      next.splice(index, 1);
      commit(next);
    },
    [tags, commit],
  );

  const clear = React.useCallback(() => commit([]), [commit]);

  return { tags, setTags, addTag, removeTag, clear };
}

export type TagsInputProps = Omit<React.ComponentProps<"div">, "onChange" | "defaultValue"> &
  Pick<
    UseTagsInputOptions,
    | "value"
    | "defaultValue"
    | "onChange"
    | "maxTags"
    | "minLength"
    | "maxLength"
    | "allowDuplicates"
    | "trim"
    | "validate"
  > & {
    placeholder?: string;
    disabled?: boolean;
    delimiters?: string[];
    onTagAdd?: (tag: string) => void;
    onTagRemove?: (tag: string, index: number) => void;
    onValidationError?: (message: string) => void;
    "aria-invalid"?: boolean;
    name?: string;
  };

export function TagsInput({
  value,
  defaultValue,
  onChange,
  maxTags,
  minLength,
  maxLength,
  allowDuplicates,
  trim,
  validate,
  placeholder = "Add a tag...",
  disabled = false,
  delimiters = ["Enter", ","],
  onTagAdd,
  onTagRemove,
  onValidationError,
  className,
  name,
  "aria-invalid": ariaInvalid,
  ...props
}: TagsInputProps) {
  const { tags, addTag, removeTag } = useTagsInput({
    value,
    defaultValue,
    onChange,
    maxTags,
    minLength,
    maxLength,
    allowDuplicates,
    trim,
    validate,
  });
  const [input, setInput] = React.useState("");
  const inputRef = React.useRef<HTMLInputElement>(null);

  const commitInput = (raw: string) => {
    if (!raw.trim()) return;
    const result = addTag(raw);
    if (result === true) {
      onTagAdd?.(trim === false ? raw : raw.trim());
      setInput("");
    } else {
      onValidationError?.(result);
    }
  };

  const handleKeyDown = (event: React.KeyboardEvent<HTMLInputElement>) => {
    if (delimiters.includes(event.key)) {
      event.preventDefault();
      commitInput(input);
      return;
    }
    if (event.key === "Backspace" && input === "" && tags.length > 0) {
      const lastTag = tags[tags.length - 1];
      removeTag(tags.length - 1);
      onTagRemove?.(lastTag!, tags.length - 1);
    }
  };

  const handlePaste = (event: React.ClipboardEvent<HTMLInputElement>) => {
    const pasted = event.clipboardData.getData("text");
    if (!pasted.includes(",") && !pasted.includes("\n")) return;
    event.preventDefault();
    const parts = pasted
      .split(/[,\n]/)
      .map((p) => p.trim())
      .filter(Boolean);
    for (const part of parts) commitInput(part);
  };

  return (
    <div
      data-slot="tags-input"
      data-disabled={disabled || undefined}
      aria-invalid={ariaInvalid}
      onClick={() => inputRef.current?.focus()}
      className={cn(
        "border-input focus-within:border-ring focus-within:ring-ring/50 aria-invalid:border-destructive aria-invalid:ring-destructive/20 dark:bg-input/30 dark:aria-invalid:border-destructive/50 dark:aria-invalid:ring-destructive/40 flex min-h-8 w-full flex-wrap items-center gap-1 rounded-lg border bg-transparent px-1.5 py-1 text-sm transition-colors outline-none focus-within:ring-3 aria-invalid:ring-3 data-[disabled]:cursor-not-allowed data-[disabled]:opacity-50",
        className,
      )}
      {...props}
    >
      {tags.map((tag, i) => (
        <Badge key={`${tag}-${i}`} variant="secondary" className="data-[icon=inline-end]:pr-1">
          <span>{tag}</span>
          {!disabled && (
            <button
              type="button"
              data-icon="inline-end"
              aria-label={`Remove ${tag}`}
              onClick={(e) => {
                e.stopPropagation();
                removeTag(i);
                onTagRemove?.(tag, i);
              }}
              className="hover:bg-foreground/10 focus-visible:ring-ring/50 inline-flex size-3.5 items-center justify-center rounded-full transition-colors focus-visible:ring-2 focus-visible:outline-none"
            >
              <X className="size-2.5" />
            </button>
          )}
        </Badge>
      ))}
      <input
        ref={inputRef}
        type="text"
        value={input}
        onChange={(e) => setInput(e.target.value)}
        onKeyDown={handleKeyDown}
        onPaste={handlePaste}
        disabled={disabled}
        placeholder={tags.length === 0 ? placeholder : undefined}
        className="placeholder:text-muted-foreground min-w-[8ch] flex-1 bg-transparent px-1 py-0.5 text-sm outline-none disabled:cursor-not-allowed"
      />
      {name && <input type="hidden" name={name} value={JSON.stringify(tags)} />}
    </div>
  );
}
```

```tsx
"use client";

import type { ReactNode, RefObject } from "react";
import { useEffect, useMemo, useState } from "react";

import { cn } from "../lib/utils";
import { HoverCard, HoverCardContent, HoverCardTrigger } from "./hover-card";

export type ScrollMinimapItem = {
  label: ReactNode;
  href: `#${string}`;
  depth?: number;
  description?: ReactNode;
};

export type ScrollMinimapProps = {
  items: ScrollMinimapItem[];
  direction?: "left" | "right";
  className?: string;
  ariaLabel?: string;
  scrollContainerRef?: RefObject<HTMLElement | null>;
  onNavigate?: (item: ScrollMinimapItem) => void;
  onItemEnter?: (item: ScrollMinimapItem) => void;
};

export function ScrollMinimap({
  items,
  direction = "left",
  className,
  ariaLabel = "Section navigation",
  scrollContainerRef,
  onNavigate,
  onItemEnter,
}: ScrollMinimapProps) {
  const itemIds = useMemo(() => items.map((item) => item.href.slice(1)), [items]);
  const [hoveredIndex, setHoveredIndex] = useState<number | null>(null);
  const activeId = useActiveSection(itemIds, scrollContainerRef);

  if (!items.length) return null;

  return (
    <nav aria-label={ariaLabel} className={cn("w-12", className)}>
      <ul
        className="flex max-h-[50dvh] cursor-pointer flex-col items-stretch py-3"
        onMouseLeave={() => setHoveredIndex(null)}
      >
        {items.map((item, index) => (
          <li key={item.href} className="flex w-full items-center">
            <HoverCard open={hoveredIndex === index}>
              <HoverCardTrigger asChild>
                <a
                  href={item.href}
                  aria-label={
                    typeof item.label === "string" ? `Go to ${item.label}` : "Go to section"
                  }
                  aria-current={item.href === `#${activeId}` ? "location" : undefined}
                  data-active={item.href === `#${activeId}`}
                  className={cn(
                    "group focus-visible:ring-ring/50 flex w-12 cursor-pointer items-center rounded-sm bg-transparent py-1 focus-visible:ring-2 focus-visible:outline-none",
                    direction === "right" && "justify-end",
                    item.depth === 3 && (direction === "left" ? "ml-2" : "mr-2"),
                    item.depth && item.depth >= 4 && (direction === "left" ? "ml-4" : "mr-4"),
                  )}
                  onClick={(event) => {
                    event.preventDefault();
                    scrollToItem(item.href, scrollContainerRef?.current);
                    onNavigate?.(item);
                  }}
                  onMouseEnter={() => {
                    setHoveredIndex(index);
                    onItemEnter?.(item);
                  }}
                  onFocus={() => {
                    setHoveredIndex(index);
                    onItemEnter?.(item);
                  }}
                  onBlur={() => setHoveredIndex(null)}
                >
                  <span
                    className={cn(
                      "bg-muted-foreground/35 block h-0.5 origin-left rounded-full",
                      "transition-[width,background-color,transform] duration-300 ease-[cubic-bezier(0.22,1,0.36,1)]",
                      "group-hover:bg-primary group-focus-visible:bg-primary group-hover:scale-y-150 group-focus-visible:scale-y-150",
                      "group-data-[active=true]:bg-primary motion-reduce:transition-none",
                      direction === "right" && "origin-right",
                      getProximityWidth(index, hoveredIndex),
                    )}
                  />
                </a>
              </HoverCardTrigger>
              <HoverCardContent
                side={direction === "left" ? "right" : "left"}
                align="center"
                sideOffset={14}
                className="w-auto max-w-64 gap-1 px-3 py-2 duration-200"
              >
                <p className="line-clamp-2 text-sm font-medium">{item.label}</p>
                <p className="text-muted-foreground text-xs">
                  {item.description ?? "Jump to this section"}
                </p>
              </HoverCardContent>
            </HoverCard>
          </li>
        ))}
      </ul>
    </nav>
  );
}

function getProximityWidth(index: number, hoveredIndex: number | null) {
  if (hoveredIndex === null) return "w-3";

  const distance = Math.abs(index - hoveredIndex);
  if (distance === 0) return "w-10";
  if (distance === 1) return "w-8";
  if (distance === 2) return "w-6";
  if (distance === 3) return "w-5";
  if (distance === 4) return "w-4";
  return "w-3";
}

function useActiveSection(itemIds: string[], scrollContainerRef?: RefObject<HTMLElement | null>) {
  const [activeId, setActiveId] = useState<string | null>(null);

  useEffect(() => {
    const headings = itemIds
      .map((id) => document.getElementById(id))
      .filter((element): element is HTMLElement => element !== null);
    if (!headings.length) return;

    const scrollContainer = scrollContainerRef?.current;
    const eventTarget = scrollContainer ?? window;
    let frame: number | null = null;

    const updateActiveSection = () => {
      const containerRect = scrollContainer?.getBoundingClientRect();
      const marker = containerRect
        ? containerRect.top + containerRect.height * 0.3
        : window.innerHeight * 0.3;
      let currentId = headings[0]?.id ?? null;

      for (const heading of headings) {
        if (heading.getBoundingClientRect().top > marker) break;
        currentId = heading.id;
      }

      const atEnd = scrollContainer
        ? scrollContainer.scrollTop + scrollContainer.clientHeight >=
          scrollContainer.scrollHeight - 2
        : window.innerHeight + window.scrollY >= document.documentElement.scrollHeight - 2;
      if (atEnd) currentId = headings.at(-1)?.id ?? currentId;

      setActiveId((previousId) => (previousId === currentId ? previousId : currentId));
    };

    const scheduleUpdate = () => {
      if (frame !== null) return;
      frame = window.requestAnimationFrame(() => {
        frame = null;
        updateActiveSection();
      });
    };

    updateActiveSection();
    eventTarget.addEventListener("scroll", scheduleUpdate, { passive: true });
    window.addEventListener("resize", scheduleUpdate);

    return () => {
      eventTarget.removeEventListener("scroll", scheduleUpdate);
      window.removeEventListener("resize", scheduleUpdate);
      if (frame !== null) window.cancelAnimationFrame(frame);
    };
  }, [itemIds, scrollContainerRef]);

  return activeId;
}

function scrollToItem(href: `#${string}`, scrollContainer?: HTMLElement | null) {
  const target = document.getElementById(href.slice(1));
  if (!target) return;

  history.pushState(null, "", href);
  const behavior = window.matchMedia("(prefers-reduced-motion: reduce)").matches
    ? "auto"
    : "smooth";

  if (!scrollContainer) {
    target.scrollIntoView({ behavior });
    return;
  }

  const targetTop =
    target.getBoundingClientRect().top -
    scrollContainer.getBoundingClientRect().top +
    scrollContainer.scrollTop;
  scrollContainer.scrollTo({ top: targetTop, behavior });
}
```

```ts
  const seekWindow = Math.max(seekMax - seekMin, 0);
  const bufferedPercent =
    seekWindow > 0 ? clamp(((media.bufferedEnd - seekMin) / seekWindow) * 100, 0, 100) : 0;

    remote.togglePaused(event.nativeEvent);

    setPendingSeek(nextTime);
    remote.seeking(nextTime);

    const bounds = event.currentTarget.getBoundingClientRect();
    if (bounds.width <= 0 || seekMax <= seekMin) return;

    const progress = clamp((event.clientX - bounds.left) / bounds.width, 0, 1);
    remote.seek(seekMin + progress * (seekMax - seekMin), event.nativeEvent);

      return;

  }

      return;

      remote.changeTextTrackMode(trackIndex, "showing");

    }

    remote.userPrefersLoopChange(!media.loop);
  }
```

```ts
    const fallbackTrackIndex = media.textTracks.findIndex(
      (track) => track.kind === "captions" || track.kind === "subtitles",
    );
    const trackIndex = preferredTrackIndex >= 0 ? preferredTrackIndex : fallbackTrackIndex;

    if (trackIndex >= 0) {
      setPreferredCaptionId(media.textTracks[trackIndex]?.id ?? null);
      remote.changeTextTrackMode(trackIndex, "showing", event.nativeEvent);
    }
  }

  function toggleLoop(): void {
    remote.userPrefersLoopChange(!media.loop);
  }

  function seekToLiveEdge(): void {
    remote.seekToLiveEdge();
  }

  function requestAirPlay(): void {
    remote.requestAirPlay();
  }

  function requestGoogleCast(): void {
    remote.requestGoogleCast();
  }

  function handleMenuOpenChange(open: boolean): void {
    if (open) remote.pauseControls();
    else remote.resumeControls();
  }

  return {
    media,
    seekMin,
    seekMax,
    seekValue,
    bufferedPercent,
    togglePlayback,
    seekBy,
    previewSeek,
    commitSeek,
    startPointerSeek,
    movePointerSeek,
    finishPointerSeek,
    changeVolume,
    toggleMuted,

    changeAudioTrack,
    changeTextTrack,
    toggleCaptions,

    seekToLiveEdge,
    requestAirPlay,
    requestGoogleCast,
    handleMenuOpenChange,
  };
}
```

```tsx
"use client";

import * as React from "react";
import { tableFeatures, useTable, type ColumnDef } from "@tanstack/react-table";
import type { z } from "zod";

import { cn } from "../../lib/utils";
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "../table";

const FEATURES = tableFeatures({});

type ColumnKey<TRow> = Extract<keyof TRow, string>;

export type ZodDataTableColumnContext<TRow, TKey extends ColumnKey<TRow>> = {
  value: TRow[TKey];
  row: TRow;
  rowIndex: number;
};

export type ZodDataTableColumn<TRow, TKey extends ColumnKey<TRow>> = {
  header?: React.ReactNode;
  hidden?: boolean;
  cell?: (context: ZodDataTableColumnContext<TRow, TKey>) => React.ReactNode;
};

export type ZodDataTableColumns<TRow> = {
  [TKey in ColumnKey<TRow>]?: ZodDataTableColumn<TRow, TKey>;
};

export type ZodDataTableProps<TSchema extends z.ZodObject> = {
  schema: TSchema;
  data: Array<z.output<TSchema>>;
  columns?: ZodDataTableColumns<z.output<TSchema>>;
  emptyMessage?: string;
  className?: string;
};

export function humanizeColumnKey(key: string): string {
  return key
    .replace(/([a-z\d])([A-Z])/g, "$1 $2")
    .replace(/[_-]+/g, " ")
    .trim()
    .replace(/\b\w/g, (character) => character.toUpperCase());
}

export function formatZodDataTableValue(value: unknown): string {
  if (value === null || value === undefined) return "—";
  if (value instanceof Date) return value.toISOString();
  if (typeof value !== "object") return String(value);

  try {
    return JSON.stringify(value);
  } catch {
    return String(value);
  }
}

function getSchemaKeys<TSchema extends z.ZodObject>(
  schema: TSchema,
): Array<ColumnKey<z.output<TSchema>>> {
  // Object.keys is the runtime boundary between the Zod shape and its inferred output keys.
  return Object.keys(schema.shape) as Array<ColumnKey<z.output<TSchema>>>;
}

function createColumn<
  TSchema extends z.ZodObject,
  TRow extends z.output<TSchema>,
  TKey extends ColumnKey<TRow>,
>(
  schema: TSchema,
  key: TKey,
  column: ZodDataTableColumn<TRow, TKey> | undefined,
): ColumnDef<typeof FEATURES, TRow, unknown> | null {
  if (column?.hidden) return null;

  const metadataTitle = schema.shape[key]?.meta()?.title;
  const header = column?.header ?? metadataTitle ?? humanizeColumnKey(key);

  return {
    accessorKey: key,
    header: typeof header === "string" ? header : () => header,
    cell: (context) => {
      const value = context.row.original[key];

      if (column?.cell) {
        return column.cell({
          value,
          row: context.row.original,
          rowIndex: context.row.index,
        });
      }

      return formatZodDataTableValue(value);
    },
  };
}

function createColumns<TSchema extends z.ZodObject>(
  schema: TSchema,
  columns: ZodDataTableColumns<z.output<TSchema>> | undefined,
): Array<ColumnDef<typeof FEATURES, z.output<TSchema>, unknown>> {
  return getSchemaKeys(schema).flatMap((key) => {
    const column = createColumn(schema, key, columns?.[key]);
    return column ? [column] : [];
```

```ts
export type TreemapRect = {
  x: number;
  y: number;
  width: number;
  height: number;
};

type WeightedItem = {
  index: number;
  area: number;
};

const EMPTY_RECT: TreemapRect = { x: 0, y: 0, width: 0, height: 0 };

export function toPercentages(values: number[]): number[] {
  const total = values.reduce((sum, value) => sum + Math.max(value, 0), 0);
  if (total <= 0) return values.map(() => 0);
  return values.map((value) => (Math.max(value, 0) / total) * 100);
}

export function squarify(values: number[], width: number, height: number): TreemapRect[] {
  const rects = values.map(() => ({ ...EMPTY_RECT }));
  const total = values.reduce((sum, value) => sum + Math.max(value, 0), 0);
  if (total <= 0 || width <= 0 || height <= 0) return rects;

  const scale = (width * height) / total;
  const items: WeightedItem[] = values
    .map((value, index) => ({ index, area: Math.max(value, 0) * scale }))
    .filter((item) => item.area > 0)
    .sort((a, b) => b.area - a.area);

  let frame = { x: 0, y: 0, width, height };
  let row: WeightedItem[] = [];
  let cursor = 0;

  while (cursor < items.length) {
    const item = items[cursor];
    if (!item) break;

    if (row.length === 0) {
      row.push(item);
      cursor += 1;
      continue;
    }

    const side = Math.min(frame.width, frame.height);
    const currentWorst = worstAspectRatio(
      row.map((entry) => entry.area),
      side,
    );
    const nextWorst = worstAspectRatio([...row.map((entry) => entry.area), item.area], side);

    if (nextWorst <= currentWorst) {
      row.push(item);
      cursor += 1;
      continue;
    }

    frame = placeRow(row, frame, rects);
    row = [];
  }

  if (row.length > 0) placeRow(row, frame, rects);

  return rects;
}

function worstAspectRatio(areas: number[], side: number): number {
  const rowArea = areas.reduce((sum, area) => sum + area, 0);
  if (rowArea <= 0 || side <= 0) return Number.POSITIVE_INFINITY;

  const thickness = rowArea / side;
  let worst = 1;

  for (const area of areas) {
    const length = area / thickness;
    if (length <= 0) return Number.POSITIVE_INFINITY;
    worst = Math.max(worst, thickness / length, length / thickness);
  }

  return worst;
}

function placeRow(row: WeightedItem[], frame: TreemapRect, rects: TreemapRect[]): TreemapRect {
  const rowArea = row.reduce((sum, item) => sum + item.area, 0);
  const horizontal = frame.width >= frame.height;
  const side = horizontal ? frame.height : frame.width;
  const thickness = rowArea / side;
  let offset = 0;

  for (const item of row) {
    const length = item.area / thickness;
    rects[item.index] = horizontal
      ? { x: frame.x, y: frame.y + offset, width: thickness, height: length }
      : { x: frame.x + offset, y: frame.y, width: length, height: thickness };
    offset += length;
  }

  return horizontal
    ? {
        x: frame.x + thickness,
        y: frame.y,
        width: Math.max(frame.width - thickness, 0),
        height: frame.height,
      }
    : {
        x: frame.x,
        y: frame.y + thickness,
        width: frame.width,
        height: Math.max(frame.height - thickness, 0),
      };
}
```

```tsx
  const reduceMotion = useReducedMotion();

  useEffect(() => {

                    {formatted}
                  </span>
                )}
              </div>
            </motion.div>
          );
        })}
    </div>
  );
}
```

```ts
  | { install: string; dev?: boolean }

  npm: "npx",
  pnpm: "pnpm dlx",

  bun: "bunx --bun",

  yarn: "yarn",
  bun: "bun run",
};

function normalize(value: string): string {
  return value
    .split("\n")
    .map((line) => line.trim().replace(/[ \t]+/g, " "))
    .join("\n")
    .trim();
}

function withVerb(verb: string, body: string): string {
  return body === "" ? verb : `${verb} ${body}`;
}

function perManager(build: (manager: PackageManager) => string): Record<PackageManager, string> {
  return {
    npm: build("npm"),
    pnpm: build("pnpm"),
    yarn: build("yarn"),
    bun: build("bun"),
  };
}

export function resolveCommands(source: CliCommandSource): Record<PackageManager, string> {
  if ("commands" in source) {
    return { ...source.commands };
  }
  if ("install" in source) {
    const packages = normalize(source.install);
    const flag = source.dev === true ? " -D" : "";
    return perManager((manager) => withVerb(`${INSTALLERS[manager]}${flag}`, packages));
  }
  if ("execute" in source) {
    const command = normalize(source.execute);
    return perManager((manager) => withVerb(EXECUTORS[manager], command));
  }
  const script = normalize(source.run);
  return perManager((manager) => withVerb(RUNNERS[manager], script));
}
```

```ts
export type ShellTokenType = "program" | "flag" | "string" | "operator" | "text" | "space";

export type ShellToken = {
  type: ShellTokenType;
  value: string;
};

const OPERATORS = new Set([";", "|"]);

function startsToken(line: string, index: number): boolean {
  const char = line[index];
  if (char === " " || char === "\t" || char === "'" || char === '"') {
    return true;
  }
  if (char !== undefined && OPERATORS.has(char)) {
    return true;
  }
  return char === "&" && line[index + 1] === "&";
}

/**
 * Splits one command line into colored spans. Concatenating every `value` gives the input back
 * byte for byte, so the rendered text never differs from the copied text.
 */
export function tokenizeCommand(line: string): ShellToken[] {
  const tokens: ShellToken[] = [];
  let index = 0;
  let nextIsProgram = true;

  while (index < line.length) {
    const char = line[index] as string;

    if (char === " " || char === "\t") {
      let value = "";
      while (index < line.length && (line[index] === " " || line[index] === "\t")) {
        value += line[index];
        index += 1;
      }
      tokens.push({ type: "space", value });
      continue;
    }

    if (char === "'" || char === '"') {
      let value = char;
      index += 1;
      while (index < line.length && line[index] !== char) {
        value += line[index];
        index += 1;
      }
      if (index < line.length) {
        value += line[index];
        index += 1;
      }
      tokens.push({ type: "string", value });
      nextIsProgram = false;
      continue;
    }

    if (OPERATORS.has(char) || (char === "&" && line[index + 1] === "&")) {
      let value = char;
      index += 1;
      if (index < line.length && line[index] === char) {
        value += line[index];
        index += 1;
      }
      tokens.push({ type: "operator", value });
      nextIsProgram = true;
      continue;
    }

    let value = "";
    while (index < line.length && !startsToken(line, index)) {
      value += line[index];
      index += 1;
    }
    const type: ShellTokenType = nextIsProgram
      ? "program"
      : value.startsWith("-")
        ? "flag"
        : "text";
    tokens.push({ type, value });
    nextIsProgram = false;
  }

  return tokens;
}
```

```ts
  window.addEventListener(CHANGE_EVENT, onChange);
  return () => {
    window.removeEventListener("storage", onChange);
    window.removeEventListener(CHANGE_EVENT, onChange);
  };
}

/**
 * The selected package manager, shared by every `CliCommand` on the page and across browser tabs.
 * The server renders the default so the first paint matches the markup.
 */
export function usePackageManager(
  defaultManager: PackageManager,
): [PackageManager, (manager: PackageManager) => void] {
  const manager = useSyncExternalStore(
    subscribe,
    () => readStored() ?? defaultManager,
    () => defaultManager,
  );

  const setManager = (next: PackageManager): void => {
    try {
      window.localStorage.setItem(STORAGE_KEY, next);
    } catch {
      // Storage can be blocked; the in-page event still updates every command.
    }
    window.dispatchEvent(new Event(CHANGE_EVENT));
  };

  return [manager, setManager];
}
```

```ts
// Ported from https://github.com/stackedlol/ascii-sharks (used with the author's permission).
export const TAU = Math.PI * 2;

export const clamp = (v: number, lo: number, hi: number) => (v < lo ? lo : v > hi ? hi : v);

export const rand = (lo: number, hi: number) => lo + Math.random() * (hi - lo);

export const randInt = (lo: number, hi: number) => Math.floor(rand(lo, hi + 1));

export function smoothstep(e0: number, e1: number, x: number) {
  const t = clamp((x - e0) / (e1 - e0), 0, 1);
  return t * t * (3 - 2 * t);
}

/** Shortest signed angle from a to b, in (-PI, PI]. */
export function angleDiff(a: number, b: number) {
  let d = (b - a) % TAU;
  if (d > Math.PI) d -= TAU;
  if (d < -Math.PI) d += TAU;
  return d;
}

/** Cheap integer hash → [0, 1). Deterministic, so textures stay stable. */
export function hash3(x: number, y: number, z: number) {
  let h = Math.imul(x | 0, 374761393) ^ Math.imul(y | 0, 668265263) ^ Math.imul(z | 0, 1442695041);
  h = Math.imul(h ^ (h >>> 13), 1274126177);
  h ^= h >>> 16;
  return (h >>> 0) / 4294967296;
}

const K1 = 1 / 260;
const K2 = 1 / 310;
const K3 = 1 / 190;
const CURRENT_SCALE = 3400;

/**
 * Divergence-free ocean current: the curl of a slowly drifting stream
 * function ψ(x, y, t). Because it has no sources or sinks, it moves the
 * schools around without bunching them up in one corner.
 */
export function current(x: number, y: number, t: number, out: { x: number; y: number }) {
  const a = x * K1 + t * 0.07;
  const b = y * K2 - t * 0.05;
  const c = (x + y) * K3 + t * 0.11;
  // u = ∂ψ/∂y, v = -∂ψ/∂x with ψ = sin(a)cos(b) + 0.5 sin(c)
  out.x = (-K2 * Math.sin(a) * Math.sin(b) + 0.5 * K3 * Math.cos(c)) * CURRENT_SCALE;
  out.y = (-K1 * Math.cos(a) * Math.cos(b) - 0.5 * K3 * Math.cos(c)) * CURRENT_SCALE;
  return out;
}
```

```ts
// Ported from https://github.com/stackedlol/ascii-sharks (used with the author's permission).
/**
 * Artificial-life sim for the hero: tiny fish schools and the sharks that hunt them.
 *
 * Fish: Reynolds boids (separation / alignment / cohesion) in Couzin-style
 * zones, with a blind spot behind each fish. Alarm spreads from fish to fish
 * through the school (a "wave of agitation"), so a whole school bolts even
 * when only its edge can see the shark. Fleeing fish break sideways around the
 * shark's path, which gives the fountain effect seen in real bait balls.
 *
 * Sharks: Reynolds steering (wander + pursuit + separation) on a
 * follow-the-leader spine. Hunger builds over time and drops when they eat.
 * They prefer fish in sparse spots, since dense schools confuse predators,
 * so stragglers get picked off. Eaten fish come back later as new schools
 * swimming in from the edges, which keeps the population steady.
 */
import { angleDiff, clamp, current, rand, randInt, TAU } from "./math";
import { pickSpecies, SPECIES } from "./species";

export const SHARK_NODES = 20;

export type SharkKind =
  "classic" | "hammerhead" | "zebra" | "tiger" | "greathammer" | "mako" | "thresher" | "nurse";

/** Which sharks to put in the water; size is relative to the biggest. */
export type SharkSpec = { kind: SharkKind; size: number };

// The landing hero: four of the same shark at different depths.
const DEFAULT_SHARKS: SharkSpec[] = [
  { kind: "classic", size: 1 },
  { kind: "classic", size: 0.7 },
  { kind: "classic", size: 0.84 },
  { kind: "classic", size: 0.77 },
];

export interface Fish {
  x: number;
  y: number;
  vx: number;
  vy: number;
  species: number;
  /** 0 calm → 1 panicking. Spreads to neighbours. */
  alarm: number;
  alive: boolean;
  respawnAt: number;
  /** Tail-beat phase. */
  phase: number;
  /** Smoothed angular velocity (rad/s); hard turns flash silver. */
  turn: number;
  /** 0.55 far → 1 near. */
  depth: number;
  /** Rotation sense shared by a school when it mills into a bait ball. */
  millDir: 1 | -1;
}

export interface Shark {
  id: number;
  kind: SharkKind;
  length: number;
  /** Spine positions, snout first: [x0, y0, x1, y1, ...]. */
  nodes: Float32Array;
  heading: number;
  speed: number;
  /** Tail-beat phase; drives the body wave in the renderer. */
  phase: number;
  /** Smoothed angular velocity (rad/s); the shark banks its fins into turns. */
  turn: number;
  hunger: number;
  wander: number;
  orbitDir: 1 | -1;
  orbitRadius: number;
  target: Fish | null;
  retargetAt: number;
  /** 0 far / dim → 1 near / bright. */
  depth: number;
}

export interface WorldEvent {
  kind: "eat" | "splash";
  x: number;
  y: number;
}

const FISH_SEPARATION_K = 420;
const FISH_ALIGN_K = 1.4;
const FISH_COHESION_K = 1;
const FISH_FLEE_K = 1100;
const BLIND_COS = -0.6; // ~127° either side of the heading is visible

export class World {
  width = 0;
  height = 0;
  time = 0;
  fish: Fish[] = [];
  sharks: Shark[] = [];
  events: WorldEvent[] = [];
  pointer = { x: 0, y: 0, active: false };

  /** Fish neighbourhood radius; the spatial hash bucket size matches it. */
  private perception = 60;
  /** Per-species personal space in px (x, y). */
  private sepX: number[] = [];
  private sepY: number[] = [];
  private foreignSep = 40;
  private bucketCols = 0;
  private bucketRows = 0;
  private bucketHead = new Int32Array(0);
  private bucketCount = new Uint16Array(0);
  private bucketNext = new Int32Array(0);
  private lastSpawn = 0;
  private tmp = { x: 0, y: 0 };

  /** fishCount overrides the area-based population (the sidebar wants a single small school). */
  constructor(
    private sharkSpecs: SharkSpec[] = DEFAULT_SHARKS,
    private fishCount?: number,
  ) {}

  resize(width: number, height: number, cellW: number, cellH: number) {
    const first = this.width === 0;
    this.width = width;
    this.height = height;
    // Spacing is set in grid cells so every fish lands in its own cell.
    this.sepX = SPECIES.map((sp) => sp.spacing[0] * cellW);
    this.sepY = SPECIES.map((sp) => sp.spacing[1] * cellH);
    this.perception = Math.max(cellH * 3, cellW * 5.5);
    this.foreignSep = cellW * 3;

    if (first || this.fish.length === 0) {
      this.seed();
      return;
    }
    for (const f of this.fish) {
      f.x = clamp(f.x, 0, width);
      f.y = clamp(f.y, 0, height);
    }
  }

  seed() {
    const { width: W, height: H } = this;
    const minSide = Math.min(W, H);
    const baseLength = clamp(minSide * 0.44, 170, 430);
    const sharkCount = this.sharkSpecs.length;

    this.sharks = [];
    for (let i = 0; i < sharkCount; i++) {
      const { kind, size } = this.sharkSpecs[i];
      const length = baseLength * size;
      const orbitRadius = minSide * rand(0.28, 0.4);
      const a = (i / sharkCount) * TAU + rand(-0.4, 0.4);
      const orbitDir: 1 | -1 = i % 2 === 0 ? 1 : -1;
      const x = W / 2 + Math.cos(a) * orbitRadius * (W / minSide) * 0.9;
      const y = H / 2 + Math.sin(a) * orbitRadius;
      const heading = a + (orbitDir * Math.PI) / 2;
      const nodes = new Float32Array(SHARK_NODES * 2);
      const seg = length / (SHARK_NODES - 1);
      for (let n = 0; n < SHARK_NODES; n++) {
        nodes[n * 2] = x - Math.cos(heading) * seg * n;
        nodes[n * 2 + 1] = y - Math.sin(heading) * seg * n;
      }
      this.sharks.push({
        id: i + 1,
        kind,
        length,
        nodes,
        heading,
        speed: 30,
        phase: rand(0, TAU),
        turn: 0,
        hunger: rand(0.25, 0.75),
        wander: 0,
        orbitDir,
        orbitRadius,
        target: null,
        retargetAt: 0,
        depth: size,
      });
    }
    // Smaller (farther) sharks draw first so the big one swims over them.
    this.sharks.sort((a, b) => a.length - b.length);

    const total = this.fishCount ?? clamp(Math.round((W * H) / 4400), 85, 340);
    this.fish = [];
    while (this.fish.length < total) {
      const species = pickSpecies();
      const sp = SPECIES[species];
      const size = Math.min(randInt(sp.schoolSize[0], sp.schoolSize[1]), total - this.fish.length);
      const millDir: 1 | -1 = Math.random() < 0.5 ? 1 : -1;
      const cx = rand(W * 0.06, W * 0.94);
      const cy = rand(H * 0.08, H * 0.92);
      // Mostly level headings: fish cruise sideways far more than up or down.
      const dir = (Math.random() < 0.5 ? 0 : Math.PI) + rand(-0.5, 0.5);
      for (let k = 0; k < size; k++) {
        this.fish.push({
          x: cx + rand(-1, 1) * this.sepX[species] * Math.sqrt(size),
          y: cy + rand(-1, 1) * this.sepY[species] * Math.sqrt(size) * 0.6,
          vx: Math.cos(dir) * sp.cruise,
          vy: Math.sin(dir) * sp.cruise,
          species,
          alarm: 0,
          alive: true,
          respawnAt: 0,
          phase: rand(0, TAU),
          turn: 0,
          depth: rand(0.55, 1),
          millDir,
        });
      }
    }
    this.bucketNext = new Int32Array(this.fish.length);
  }

  splash(x: number, y: number) {
    this.events.push({ kind: "splash", x, y });
    const r = 240;
    for (const f of this.fish) {
      if (!f.alive) continue;
      const dx = f.x - x;
      const dy = f.y - y;
      const d = Math.hypot(dx, dy);
      if (d > r || d < 1e-3) continue;
      // Flash expansion: everybody nearby bolts straight out.
      const k = (1 - d / r) * 260;
      f.vx += (dx / d) * k;
      f.vy += (dy / d) * k;
      f.alarm = 1;
    }
  }

  step(dt: number) {
    this.time += dt;
    this.rebuildHash();
    this.stepFish(dt);
    this.stepSharks(dt);
    this.respawn();
  }

  private rebuildHash() {
    const size = this.perception;
    const pad = 2;
    this.bucketCols = Math.ceil(this.width / size) + pad * 2;
    this.bucketRows = Math.ceil(this.height / size) + pad * 2;
    const n = this.bucketCols * this.bucketRows;
    if (this.bucketHead.length !== n) {
      this.bucketHead = new Int32Array(n);
      this.bucketCount = new Uint16Array(n);
    }
    this.bucketHead.fill(-1);
    this.bucketCount.fill(0);
    if (this.bucketNext.length !== this.fish.length) {
      this.bucketNext = new Int32Array(this.fish.length);
    }
    for (let i = 0; i < this.fish.length; i++) {
      const f = this.fish[i];
      if (!f.alive) continue;
      const b = this.bucketOf(f.x, f.y);
      this.bucketNext[i] = this.bucketHead[b];
      this.bucketHead[b] = i;
      this.bucketCount[b]++;
    }
  }

  private bucketOf(x: number, y: number) {
    const size = this.perception;
    const c = clamp(Math.floor(x / size) + 2, 0, this.bucketCols - 1);
    const r = clamp(Math.floor(y / size) + 2, 0, this.bucketRows - 1);
    return r * this.bucketCols + c;
  }

  private stepFish(dt: number) {
    const { fish, sharks, pointer, width: W, height: H, time } = this;
    const R = this.perception;
    const R2 = R * R;
    const foreignSep = this.foreignSep;
    const size = this.perception;
    const cols = this.bucketCols;
    const rows = this.bucketRows;
    const head = this.bucketHead;
    const next = this.bucketNext;
    const cur = this.tmp;
    const calmDecay = Math.exp(-dt * 0.7);
    const contagion = Math.min(1, dt * 6);

    for (let i = 0; i < fish.length; i++) {
      const f = fish[i];
      if (!f.alive) continue;
      const sp = SPECIES[f.species];
      // Alarmed fish want more room: the school flashes apart around a
      // shark and closes back up once the threat fades.
      const spread = 1 + 1.8 * f.alarm;
      const sepX = this.sepX[f.species] * spread;
      const sepY = this.sepY[f.species] * spread;

      const speed = Math.hypot(f.vx, f.vy) || 1;
      const hx = f.vx / speed;
      const hy = f.vy / speed;

      let ax = 0;
      let ay = 0;
      let sx = 0;
      let sy = 0;
      let avx = 0;
      let avy = 0;
      let cx = 0;
      let cy = 0;
      let n = 0;
      let neighbourAlarm = 0;

      const bc = clamp(Math.floor(f.x / size) + 2, 0, cols - 1);
      const br = clamp(Math.floor(f.y / size) + 2, 0, rows - 1);
      for (let r = Math.max(0, br - 1); r <= Math.min(rows - 1, br + 1); r++) {
        for (let c = Math.max(0, bc - 1); c <= Math.min(cols - 1, bc + 1); c++) {
          let j = head[r * cols + c];
          while (j !== -1) {
            if (j !== i) {
              const o = fish[j];
              const dx = o.x - f.x;
              const dy = o.y - f.y;
              const d2 = dx * dx + dy * dy;
              if (d2 < R2 && d2 > 1e-6) {
                const d = Math.sqrt(d2);
                // Blind zone directly behind the fish.
                if ((dx * hx + dy * hy) / d > BLIND_COS) {
                  // Elliptical personal space: wide along x, tight along y.
                  const qx = dx / sepX;
                  const qy = dy / sepY;
                  const q2 = qx * qx + qy * qy;
                  if (q2 < 1) {
                    const q = Math.sqrt(q2) || 1e-3;
                    const k = (1 - q) / q;
                    sx -= qx * k;
                    sy -= qy * k;
                  }
                  if (o.species === f.species) {
                    avx += o.vx;
                    avy += o.vy;
                    cx += dx;
                    cy += dy;
                    n++;
                  } else if (d < foreignSep) {
                    // Different species keep their own schools.
                    const k = ((1 - d / foreignSep) / d) * 0.6;
                    sx -= dx * k;
                    sy -= dy * k;
                  }
                  if (o.alarm > neighbourAlarm) neighbourAlarm = o.alarm;
                }
              }
            }
            j = next[j];
          }
        }
      }

      ax += sx * FISH_SEPARATION_K;
      ay += sy * FISH_SEPARATION_K;
      if (n > 0) {
        const alignK = FISH_ALIGN_K * sp.align;
        const cohesionK = FISH_COHESION_K * sp.cohesion * (1 - 0.8 * f.alarm);
        ax += (avx / n - f.vx) * alignK + (cx / n) * cohesionK;
        ay += (avy / n - f.vy) * alignK + (cy / n) * cohesionK;
        if (sp.mill > 0) {
          // Circle the local centre: the school spins into a bait ball.
          const cl = Math.hypot(cx, cy) || 1;
          ax += (-cy / cl) * f.millDir * sp.mill * (1 - f.alarm);
          ay += (cx / cl) * f.millDir * sp.mill * (1 - f.alarm);
        }
      }

      // Predators. Distance is measured to the nearest point along the front
      // half of the shark, so fish also give the flanks a wide berth.
      let fear = 0;
      for (const sh of sharks) {
        const fearR = sh.length * 0.6 + 60;
        let best = Infinity;
        let bx = 0;
        let by = 0;
        for (let k = 0; k <= 12; k += 3) {
          const nx = sh.nodes[k * 2];
          const ny = sh.nodes[k * 2 + 1];
          const d2 = (f.x - nx) ** 2 + (f.y - ny) ** 2;
          if (d2 < best) {
            best = d2;
            bx = nx;
            by = ny;
          }
        }
        const d = Math.sqrt(best);
        if (d >= fearR || d < 1e-3) continue;
        let w = 1 - d / fearR;
        w *= w;
        const awayX = (f.x - bx) / d;
        const awayY = (f.y - by) / d;
        // Fountain effect: break to whichever side of the shark's path you're on.
        const shx = Math.cos(sh.heading);
        const shy = Math.sin(sh.heading);
        const side = shx * (f.y - sh.nodes[1]) - shy * (f.x - sh.nodes[0]) >= 0 ? 1 : -1;
        ax += (awayX * 0.7 - shy * side * 0.9) * FISH_FLEE_K * w;
        ay += (awayY * 0.7 + shx * side * 0.9) * FISH_FLEE_K * w;
        fear = Math.max(fear, Math.sqrt(w));
      }

      if (pointer.active) {
        const pr = 130;
        const dx = f.x - pointer.x;
        const dy = f.y - pointer.y;
        const d = Math.hypot(dx, dy);
        if (d < pr && d > 1e-3) {
          const w = (1 - d / pr) ** 2;
          ax += (dx / d) * FISH_FLEE_K * w;
          ay += (dy / d) * FISH_FLEE_K * w;
          fear = Math.max(fear, Math.sqrt(w) * 0.8);
        }
      }

      // Alarm: direct threat sets it, neighbours pass it along (slightly
      // weaker each hop), and it fades on its own.
      let alarm = Math.max(f.alarm * calmDecay, fear);
      const relayed = neighbourAlarm * 0.86;
      if (relayed > alarm) alarm += (relayed - alarm) * contagion;
      f.alarm = alarm;

      // Soft walls.
      const m = 36;
      if (f.x < m) ax += (m - f.x) * 7;
      else if (f.x > W - m) ax -= (f.x - (W - m)) * 7;
      if (f.y < m) ay += (m - f.y) * 7;
      else if (f.y > H - m) ay -= (f.y - (H - m)) * 7;

      // Ambient current and a little individual restlessness.
      current(f.x, f.y, time, cur);
      ax += cur.x * 0.5 + (Math.random() - 0.5) * sp.jitter;
      ay += cur.y * 0.5 + (Math.random() - 0.5) * sp.jitter;
      // Swim level, except when bolting from danger.
      ay -= f.vy * sp.verticalDrag * (1 - 0.7 * f.alarm);

      // Relax toward a preferred speed that rises with panic.
      const preferred = sp.cruise * (1 + 1.3 * f.alarm);
      ax += (preferred - speed) * hx * 1.6;
      ay += (preferred - speed) * hy * 1.6;

      const before = Math.atan2(f.vy, f.vx);
      f.vx += ax * dt;
      f.vy += ay * dt;
      const maxSpeed = sp.maxSpeed * (1 + 1.8 * f.alarm);
      const s = Math.hypot(f.vx, f.vy);
      if (s > maxSpeed) {
        f.vx *= maxSpeed / s;
        f.vy *= maxSpeed / s;
      } else if (s < sp.minSpeed && s > 1e-3) {
        f.vx *= sp.minSpeed / s;
        f.vy *= sp.minSpeed / s;
      }
      f.turn += (angleDiff(before, Math.atan2(f.vy, f.vx)) / dt - f.turn) * Math.min(1, dt * 8);
      f.phase += dt * (7 + Math.hypot(f.vx, f.vy) * 0.12);
      f.x = clamp(f.x + f.vx * dt, -60, W + 60);
      f.y = clamp(f.y + f.vy * dt, -60, H + 60);
    }
  }

  private stepSharks(dt: number) {
    const { sharks, width: W, height: H, time } = this;
    const minSide = Math.min(W, H);
    const cxW = W / 2;
    const cyW = H / 2;

    for (const sh of sharks) {
      const L = sh.length;
      const nodes = sh.nodes;
      const hx = nodes[0];
      const hy = nodes[1];

      sh.hunger = Math.min(1, sh.hunger + dt * 0.03);

      // Momentum.
      let dx = Math.cos(sh.heading);
      let dy = Math.sin(sh.heading);

      // Wander: a slowly drifting offset on a circle ahead of the snout.
      sh.wander = clamp(sh.wander + (Math.random() - 0.5) * dt * 3.2, -1.1, 1.1);
      dx += Math.cos(sh.heading + sh.wander) * 0.45;
      dy += Math.sin(sh.heading + sh.wander) * 0.45;

      // Hunt.
      if (time >= sh.retargetAt) {
        sh.target = this.findPrey(sh);
        sh.retargetAt = time + 0.45;
      }
      const t = sh.target;
      let hunting = false;
      let closing = 0;
      if (t && t.alive && sh.hunger > 0.3) {
        // Pursuit: aim at where the fish is about to be.
        const tx = t.x + t.vx * 0.35 - hx;
        const ty = t.y + t.vy * 0.35 - hy;
        const d = Math.hypot(tx, ty) || 1;
        const w = 1.4 + 2.6 * sh.hunger;
        dx += (tx / d) * w;
        dy += (ty / d) * w;
        hunting = true;
        closing = clamp(1 - d / (L * 0.9), 0, 1);
      } else {
        // Patrol: a lazy circuit around the headline.
        const rx = (hx - cxW) / (W / minSide);
        const ry = hy - cyW;
        const r = Math.hypot(rx, ry) || 1;
        const radius = sh.orbitRadius * (1 + 0.15 * Math.sin(time * 0.05 + sh.id));
        const radial = clamp((radius - r) / radius, -1, 1);
        dx += ((-ry / r) * sh.orbitDir + (rx / r) * radial * 0.9) * 0.55;
        dy += ((rx / r) * sh.orbitDir + (ry / r) * radial * 0.9) * 0.55;
      }

      // Give other sharks room.
      for (const other of sharks) {
        if (other === sh) continue;
        for (let k = 0; k < SHARK_NODES; k += 4) {
          const ox = hx - other.nodes[k * 2];
          const oy = hy - other.nodes[k * 2 + 1];
          const d = Math.hypot(ox, oy);
          const range = (L + other.length) * 0.45;
          if (d < range && d > 1e-3) {
            const w = (1 - d / range) * 3;
            dx += (ox / d) * w;
            dy += (oy / d) * w;
          }
        }
      }

      // Keep the headline mostly clear: a soft elliptical no-swim zone.
      const ex = (hx - cxW) / (W * 0.3);
      const ey = (hy - cyW) / (H * 0.16);
      const e2 = ex * ex + ey * ey;
      if (e2 < 1) {
        const e = Math.sqrt(e2) || 1e-3;
        const w = (1 - e) * (hunting ? 1 : 2.5);
        dx += (ex / e) * w;
        dy += (ey / e) * w;
      }

      // Turn back once the snout leaves the frame, so bodies dip off-screen
      // now and then but the sharks never wander away for good.
      const over = Math.max(-hx, hx - W, -hy, hy - H);
      if (over > -L * 0.05) {
        const w = clamp((over + L * 0.05) / (L * 0.35), 0, 1) * 3.5;
        const toX = cxW - hx;
        const toY = cyW - hy;
        const d = Math.hypot(toX, toY) || 1;
        dx += (toX / d) * w;
        dy += (toY / d) * w;
      }

      const desired = Math.atan2(dy, dx);
      // Sharks can't pivot on the spot: the turning circle grows with speed.
      const agility = Math.min(hunting ? 1.7 + closing : 0.8, sh.speed / (L * 0.22));
      const maxTurn = agility * dt;
      const turned = clamp(angleDiff(sh.heading, desired), -maxTurn, maxTurn);
      sh.heading += turned;
      sh.turn += (turned / dt - sh.turn) * Math.min(1, dt * 3);

      const cruise = 30 + L * 0.04;
      const targetSpeed = hunting ? cruise * (1.45 + closing * 1.5) : cruise;
      sh.speed += (targetSpeed - sh.speed) * Math.min(1, dt * 1.1);
      sh.phase += dt * (2 + sh.speed / 24);

      nodes[0] += Math.cos(sh.heading) * sh.speed * dt;
      nodes[1] += Math.sin(sh.heading) * sh.speed * dt;

      // Follow-the-leader spine, with a stiffness limit per joint so the
      // body bends into a C at most and never folds back on itself.
      const seg = L / (SHARK_NODES - 1);
      const maxBend = 0.16;
      let prevAngle = sh.heading + Math.PI;
      for (let k = 1; k < SHARK_NODES; k++) {
        const px = nodes[(k - 1) * 2];
        const py = nodes[(k - 1) * 2 + 1];
        let angle = Math.atan2(nodes[k * 2 + 1] - py, nodes[k * 2] - px);
        angle = prevAngle + clamp(angleDiff(prevAngle, angle), -maxBend, maxBend);
        nodes[k * 2] = px + Math.cos(angle) * seg;
        nodes[k * 2 + 1] = py + Math.sin(angle) * seg;
        prevAngle = angle;
      }

      // Feed.
      if (sh.hunger > 0.15) {
        const mouthR = L * 0.05 + 8;
        const mx = nodes[0];
        const my = nodes[1];
        for (const f of this.fish) {
          if (!f.alive) continue;
          if ((f.x - mx) ** 2 + (f.y - my) ** 2 < mouthR * mouthR) {
            f.alive = false;
            f.respawnAt = time + rand(3, 8);
            sh.hunger = Math.max(0, sh.hunger - 0.16);
            this.events.push({ kind: "eat", x: f.x, y: f.y });
          }
        }
      }
    }
  }

  /**
   * Pick prey in the shark's forward cone, preferring close fish in sparse
   * buckets (stragglers) over fish buried in a dense school.
   */
  private findPrey(sh: Shark): Fish | null {
    const range = sh.length * 0.9 + 160;
    const hx = sh.nodes[0];
    const hy = sh.nodes[1];
    const fx = Math.cos(sh.heading);
    const fy = Math.sin(sh.heading);
    let best: Fish | null = null;
    let bestScore = Infinity;
    let currentScore = Infinity;

    for (const f of this.fish) {
      if (!f.alive) continue;
      const dx = f.x - hx;
      const dy = f.y - hy;
      const d = Math.hypot(dx, dy);
      if (d > range || d < 1e-3) continue;
      const cos = (dx * fx + dy * fy) / d;
      if (cos < 0.2) continue;
      const crowd = this.bucketCount[this.bucketOf(f.x, f.y)];
      const score = d * (1 + (1 - cos) * 0.8) * (1 + crowd * 0.07);
      if (score < bestScore) {
        bestScore = score;
        best = f;
      }
      if (f === sh.target) currentScore = score;
    }
    // Stick with the current target unless something is clearly better.
    if (sh.target && currentScore < bestScore * 1.3) return sh.target;
    return best;
  }

  private respawn() {
    const { time, width: W, height: H } = this;
    const due: Fish[] = [];
    for (const f of this.fish) if (!f.alive && f.respawnAt <= time) due.push(f);
    if (due.length === 0) return;
    if (due.length < 6 && time - this.lastSpawn < 9) return;
    this.lastSpawn = time;

    // A new school swims in from a random edge.
    const side = randInt(0, 3);
    const along = rand(0.15, 0.85);
    const x = side === 0 ? -20 : side === 1 ? W + 20 : along * W;
    const y = side === 2 ? -20 : side === 3 ? H + 20 : along * H;
    const inward = Math.atan2(H / 2 - y, W / 2 - x) + rand(-0.5, 0.5);
    const species = pickSpecies();
    const sp = SPECIES[species];
    const millDir: 1 | -1 = Math.random() < 0.5 ? 1 : -1;
    for (const f of due.slice(0, sp.schoolSize[1])) {
      f.alive = true;
      f.species = species;
      f.alarm = 0;
      f.turn = 0;
      f.millDir = millDir;
      f.x = x + rand(-1, 1) * this.perception;
      f.y = y + rand(-1, 1) * this.perception;
      f.vx = Math.cos(inward) * sp.cruise * 1.4;
      f.vy = Math.sin(inward) * sp.cruise * 1.4;
    }
  }
}
```

```ts
  // that's as long as the shark itself. Big eyes, long sickle pectorals, bronze-violet skin.
  thresher: {
    halfWidth: (s) => {
      if (s < 0.08) return 0.07 * Math.pow(Math.sin(((s / 0.08) * Math.PI) / 2), 0.75);
      if (s < 0.2) return 0.07 + 0.02 * smoothstep(0.08, 0.2, s);
      if (s < 0.48) return 0.012 + 0.078 * Math.pow(1 - (s - 0.2) / 0.28, 1.3);
      // The whip: thin, with a slight sickle swell before it tapers out.
      const k = (s - 0.48) / 0.52;
      return 0.012 * (1 - k) + 0.006 * Math.sin(k * Math.PI * 0.9) + 0.002;
    },
    pectoral: [0.12, 0.27, 0.36, 0.2],
    falcate: true,
    pelvic: [0.37, 0.41, 0.44, 0.045],
    caudal: 0,
    dorsal: [0.24, 0.32],
    secondDorsal: [0.45, 0.47],
    eye: [0.05, 0.1],
    gills: [0.11, 0.18],
    tint: [1.0, 0.9, 1.08],
  },
  // Nurse shark: a blunt, broad head, a heavy flattened body with both dorsals set far back,
  // rounded paddle pectorals, and a long low tail with no lower lobe. Tawny brown.
  nurse: {
    halfWidth: (s) => {
      if (s < 0.08) return 0.085 * Math.pow(Math.sin(((s / 0.08) * Math.PI) / 2), 0.4);
      if (s < 0.3) return 0.085 + 0.02 * smoothstep(0.08, 0.3, s);
      if (s < 0.72) return 0.022 + 0.083 * Math.pow(1 - (s - 0.3) / 0.42, 1.1);
      const k = (s - 0.72) / 0.28;
      return 0.022 * (1 - k * k) + 0.014 * Math.sin(k * Math.PI);
    },
    pectoral: [0.16, 0.34, 0.4, 0.17],
    falcate: false,
    pelvic: [0.45, 0.52, 0.55, 0.075],
    caudal: 0,
    dorsal: [0.5, 0.58],
    secondDorsal: [0.63, 0.69],
    eye: [0.05, 0.075],
    gills: [0.14, 0.22],
    tint: [1.1, 0.96, 0.76],
  },
};

const KINDS = Object.keys(SHAPES) as SharkKind[];

function inTriangle(
  px: number,
  py: number,
  ax: number,
  ay: number,
  bx: number,
  by: number,
  cx: number,
  cy: number,
) {
  const d1 = (px - bx) * (ay - by) - (ax - bx) * (py - by);
  const d2 = (px - cx) * (by - cy) - (bx - cx) * (py - cy);
  const d3 = (px - ax) * (cy - ay) - (cx - ax) * (py - ay);
  const neg = d1 < 0 || d2 < 0 || d3 < 0;
  const pos = d1 > 0 || d2 > 0 || d3 > 0;
  return !(neg && pos);
}

/** Cell grid, snapped to device pixels, stroked in `stroke` (a CSS colour). */
export function paintGrid(
  g: CanvasRenderingContext2D,
  W: number,
  H: number,
  cw: number,
  ch: number,
  dpr: number,
  stroke: string,
) {
  g.setTransform(1, 0, 0, 1, 0, 0);
  g.strokeStyle = stroke;
  g.lineWidth = 1;
  g.beginPath();
  for (let c = 0; c <= Math.ceil(W / cw); c++) {
    const x = Math.round(c * cw * dpr) + 0.5;
    g.moveTo(x, 0);
    g.lineTo(x, H * dpr);
  }
  for (let r = 0; r <= Math.ceil(H / ch); r++) {
    const y = Math.round(r * ch * dpr) + 0.5;
    g.moveTo(0, y);
    g.lineTo(W * dpr, y);
  }
  g.stroke();
}

export class AsciiRenderer {
  private ctx: CanvasRenderingContext2D;
  private bg: HTMLCanvasElement;
  private dpr = 1;
  private width = 0;
  private height = 0;
  private cw = 12;
  private ch = 18;
  private cols = 0;
  private rows = 0;
  private fontFamily = "ui-monospace, Menlo, monospace";

  private dotKind = new Uint8Array(0); // 0 none, 1 blue, 2 green, 3 red, 4 pale
  private dotPhase = new Float32Array(0);
  private dotFreq = new Float32Array(0);
  private energy = new Float32Array(0);
  private energyTmp = new Float32Array(0);
  private heat = new Float32Array(0);
  private bucketOf = new Uint8Array(0);
  private bucketOrder = new Int32Array(0);
  private bucketCounts = new Int32Array(DOT_RGB.length * ALPHA_LEVELS + 1);
  private dotStyles: string[] = [];

  private sharkFlag = new Uint8Array(0);
  private sharkShade = new Float32Array(0);
  private sharkKey = new Uint32Array(0);
  private sharkDepth = new Float32Array(0);
  private sharkKind = new Uint8Array(0);
  /** Cells darkened by a shark's shadow on the seabed. */
  private sharkShadow = new Uint8Array(0);
  private spine = new Float32Array(SHARK_NODES * 2);
  private arc = new Float32Array(SHARK_NODES);
  private arcTotal = 1;
  private shape = SHAPES.classic;
  private sharkId = 0;
  private finTipLeft = 0;
  private finTipRight = 0;
  // Scratch outputs of sampleShark(), to avoid allocating per sample.
  private hitFlag = 0;
  private hitShade = 0;
  private hitS = 0;
  private hitV = 0;
  private hitDist = 0;

  /** Cells holding a fish this frame (frame stamp); dots skip them. */
  private fishStamp = new Uint32Array(0);
  private frameId = 0;
  /** Fish cells bucketed by brightness, so each bucket is one fillStyle. */
  private fishBuckets: number[][] = FISH_ALPHAS.map(() => []);
  private fishStyles: string[] = [];
  private palette: OceanPalette = DEFAULT_PALETTE;

  constructor(private canvas: HTMLCanvasElement) {
    const ctx = canvas.getContext("2d", { alpha: false });
    if (!ctx) throw new Error("2D canvas unavailable");
    this.ctx = ctx;
    this.bg = document.createElement("canvas");
    this.buildStyles();
  }

  /** Swap in the page colours: rebuilds the ink styles and repaints the background. */
  setPalette(p: OceanPalette) {
    this.palette = p;
    this.buildStyles();
    if (this.bg.width > 0) this.paintBackground();
  }

  private buildStyles() {
    const { foreground, mode } = this.palette;
    this.fishStyles = FISH_ALPHAS.map((a) => `rgba(${foreground.join(",")},${a})`);
    this.dotStyles = [];
    for (let c = 0; c < DOT_RGB.length; c++) {
      const [r, g, b] = c === WAKE && mode === "light" ? WAKE_RGB_LIGHT : DOT_RGB[c];
      for (let l = 0; l < ALPHA_LEVELS; l++) {
        const a = (DOT_BASE_ALPHA[c] * (l + 1)) / ALPHA_LEVELS;
        this.dotStyles.push(`rgba(${r},${g},${b},${a.toFixed(3)})`);
      }
    }
  }

  /** Greys invert on light backgrounds, so the sharks read as a negative. */
  private tone(level: number) {
    return this.palette.mode === "dark" ? level : 255 - level;
  }

  get cellWidth() {
    return this.cw;
  }

  get cellHeight() {
    return this.ch;
  }

  setFontFamily(family: string) {
    if (family) this.fontFamily = family;
  }

  resize(width: number, height: number, dpr: number) {
    this.width = width;
    this.height = height;
    this.dpr = dpr;
    this.cw = clamp(Math.round(width / 124), 8, 13);
    this.ch = Math.round(this.cw * 1.45);
    this.cols = Math.ceil(width / this.cw);
    this.rows = Math.ceil(height / this.ch);

    this.canvas.width = Math.round(width * dpr);
    this.canvas.height = Math.round(height * dpr);
    this.bg.width = this.canvas.width;
    this.bg.height = this.canvas.height;

    const n = this.cols * this.rows;
    this.dotKind = new Uint8Array(n);
    this.dotPhase = new Float32Array(n);
    this.dotFreq = new Float32Array(n);
    this.energy = new Float32Array(n);
    this.energyTmp = new Float32Array(n);
    this.heat = new Float32Array(n);
    this.bucketOf = new Uint8Array(n);
    this.bucketOrder = new Int32Array(n);
    this.sharkFlag = new Uint8Array(n);
    this.sharkShade = new Float32Array(n);
    this.sharkKey = new Uint32Array(n);
    this.sharkDepth = new Float32Array(n);
    this.sharkKind = new Uint8Array(n);
    this.sharkShadow = new Uint8Array(n);
    this.fishStamp = new Uint32Array(n);

    for (let i = 0; i < n; i++) {
      const col = i % this.cols;
      const row = (i / this.cols) | 0;
      // No ambient dots: the grid stays black. Only the cyan wake lights cells.
      this.dotKind[i] = 0;
      this.dotPhase[i] = hash3(col, row, 23) * Math.PI * 2;
      this.dotFreq[i] = 0.3 + hash3(col, row, 37) * 1.1;
    }
    this.paintBackground();
  }

  private paintBackground() {
    const g = this.bg.getContext("2d");
    if (!g) return;
    const { width: W, height: H, dpr, cw, ch } = this;
    const { background, foreground, mode } = this.palette;
    g.setTransform(dpr, 0, 0, dpr, 0, 0);
    g.fillStyle = `rgb(${background.join(",")})`;
    g.fillRect(0, 0, W, H);
    paintGrid(
      g,
      W,
      H,
      cw,
      ch,
      dpr,
      `rgba(${foreground.join(",")},${mode === "dark" ? 0.055 : 0.07})`,
    );
  }

  render(world: World, dt: number) {
    const { ctx, dpr, cols, ch } = this;
    if (cols === 0) return;
    const t = world.time;

    this.consumeEvents(world);
    this.rasterizeSharks(world.sharks, t);
    this.depositWake(world);
    this.diffuse(dt);

    ctx.setTransform(1, 0, 0, 1, 0, 0);
    ctx.drawImage(this.bg, 0, 0);
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

    this.markFish(world);
    this.drawDots(t);

    ctx.font = `600 ${Math.round(ch * 0.74)}px ${this.fontFamily}`;
    ctx.textAlign = "center";
    ctx.textBaseline = "middle";

    this.drawHeat();
    this.drawFish();
    this.drawSharks(t);
  }

  // ── simulation → field ──────────────────────────────────────────────

  private cellIndex(x: number, y: number) {
    const c = Math.floor(x / this.cw);
    const r = Math.floor(y / this.ch);
    if (c < 0 || r < 0 || c >= this.cols || r >= this.rows) return -1;
    return r * this.cols + c;
  }

  private stamp(field: Float32Array, x: number, y: number, radius: number, amount: number) {
    const c0 = Math.floor(x / this.cw);
    const r0 = Math.floor(y / this.ch);
    for (let r = r0 - radius; r <= r0 + radius; r++) {
      for (let c = c0 - radius; c <= c0 + radius; c++) {
        if (c < 0 || r < 0 || c >= this.cols || r >= this.rows) continue;
        const d = Math.hypot(c - c0, (r - r0) * 1.4);
        if (d > radius + 0.5) continue;
        const i = r * this.cols + c;
        field[i] = Math.min(1.6, field[i] + amount * (1 - d / (radius + 1)));
      }
    }
  }

  private consumeEvents(world: World) {
    for (const e of world.events) {
      if (e.kind === "eat") {
        this.stamp(this.heat, e.x, e.y, 2, 1);
        this.stamp(this.energy, e.x, e.y, 3, 0.9);
      } else {
        this.stamp(this.energy, e.x, e.y, 6, 1.4);
      }
    }
    world.events.length = 0;
  }

  private depositWake(world: World) {
    const { energy } = this;
    for (const f of world.fish) {
      if (!f.alive) continue;
      const i = this.cellIndex(f.x, f.y);
      if (i >= 0) energy[i] = Math.min(1.6, energy[i] + 0.012 + f.alarm * 0.03);
    }
    for (const sh of world.sharks) {
      // The tail sweeps the most water.
      for (let k = SHARK_NODES - 6; k < SHARK_NODES; k++) {
        const i = this.cellIndex(sh.nodes[k * 2], sh.nodes[k * 2 + 1]);
        if (i >= 0) energy[i] = Math.min(1.6, energy[i] + 0.05 + sh.speed * 0.0012);
      }
    }
    if (world.pointer.active) {
      this.stamp(energy, world.pointer.x, world.pointer.y, 1, 0.08);
    }
  }

  /** Blur + decay so wakes spread out like ripples and fade. */
  private diffuse(dt: number) {
    const { cols, rows, energy, energyTmp: tmp, heat } = this;
    const decay = Math.exp(-dt * 1.5);
    for (let r = 0; r < rows; r++) {
      for (let c = 0; c < cols; c++) {
        const i = r * cols + c;
        const l = c > 0 ? energy[i - 1] : energy[i];
        const rt = c < cols - 1 ? energy[i + 1] : energy[i];
        const u = r > 0 ? energy[i - cols] : energy[i];
        const d = r < rows - 1 ? energy[i + cols] : energy[i];
        tmp[i] = (energy[i] * 4 + l + rt + u + d) * 0.125 * decay;
      }
    }
    this.energy = tmp;
    this.energyTmp = energy;
    const heatDecay = Math.exp(-dt * 2.2);
    for (let i = 0; i < heat.length; i++) heat[i] *= heatDecay;
  }

  // ── sharks ──────────────────────────────────────────────────────────

  /**
   * Classify one point against the current shark (spine/arc already built).
   * Body coordinates: s along the spine (0 snout → 1 tail), v across it.
   * Writes hit* fields and returns whether the point is on the shark.
   */
  private sampleShark(px: number, py: number, L: number) {
    const { spine, arc } = this;
    const N = SHARK_NODES;
    let best = Infinity;
    let bestS = 0;
    let bestSign = 1;
    let bestNx = 0;
    let bestNy = 0;
    for (let k = 0; k < N - 1; k++) {
      const ax = spine[k * 2];
      const ay = spine[k * 2 + 1];
      const sx = spine[(k + 1) * 2] - ax;
      const sy = spine[(k + 1) * 2 + 1] - ay;
      const len2 = sx * sx + sy * sy || 1;
      const u = clamp(((px - ax) * sx + (py - ay) * sy) / len2, 0, 1);
      const qx = ax + sx * u;
      const qy = ay + sy * u;
      const d2 = (px - qx) ** 2 + (py - qy) ** 2;
      if (d2 < best) {
        best = d2;
        const len = Math.sqrt(len2);
        bestS = (arc[k] + u * len) / this.arcTotal;
        bestSign = sx * (py - ay) - sy * (px - ax) >= 0 ? 1 : -1;
        bestNx = (-sy / len) * bestSign;
        bestNy = (sx / len) * bestSign;
      }
    }

    const a = Math.sqrt(best);
    const s = bestS;
    const u = s * L;
    const shape = this.shape;
    const w = shape.halfWidth(s) * L;
    this.hitDist = a;
    this.hitS = s;
    this.hitV = a * bestSign;

    const head = this.shape.head;
    if (head && u < L * (head.sweep + head.depth + head.notch)) {
      // Cephalofoil: a wide wing across the snout, its tips swept back toward the eyes,
      // and on a great hammerhead a shallow notch in the middle of the front edge.
      const span = L * head.span;
      const r = a / span;
      const front = L * head.sweep * r * r + L * head.notch * Math.max(0, 1 - r / 0.14);
      // Full depth almost to the tips, so the wing reads as a solid bar rather than a thin line.
      const depth = L * head.depth * (1 - 0.2 * r * r * r * r);
      if (r <= 1 && u >= front && u <= front + depth) {
        this.hitFlag = r > 0.84 ? FLAG_EYE : FLAG_BODY;
        this.hitShade = r > 0.84 ? 0 : 0.62 - 0.2 * r + (u < front + L * 0.012 ? 0.1 : 0);
        return true;
      }
    }

    if (a <= w) {
      const across = a / (w || 1);
      // Rounded back lit from the top-left; flanks fall off into shadow.
      const facing = (bestNx * -0.6 + bestNy * -0.8) * across;
      let shade = 0.58 - across * across * 0.42 + facing * 0.26;
      let flag = FLAG_BODY;
      if (s > shape.dorsal[0] && s < shape.dorsal[1] && across < 0.3) {
        // First dorsal fin, catching the light.
        flag = FLAG_RIDGE;
        shade = 1;
      } else if (s > shape.secondDorsal[0] && s < shape.secondDorsal[1] && across < 0.4) {
        shade = Math.max(shade, 0.82);
      } else if (shape.eye && s > shape.eye[0] && s < shape.eye[1] && across > 0.6) {
        flag = FLAG_EYE;
        shade = 0;
      } else if (s > shape.gills[0] && s < shape.gills[1] && across > 0.58) {
        // Five gill slits.
        if (Math.floor(((s - shape.gills[0]) / (shape.gills[1] - shape.gills[0])) * 5) % 2 === 0)
          shade *= 0.45;
      } else if (s < 0.05) {
        // Pale snout tip.
        shade = Math.max(shade, 0.7);
      } else if (
        shape === SHAPES.zebra &&
        s > 0.08 &&
        s < 0.55 &&
        (across < 0.1 || Math.abs(across - 0.6) < 0.08)
      ) {
        // Zebra sharks carry pale ridges down the back.
        shade = Math.max(shade, 0.8);
      } else if (this.marked(s, a * bestSign, L, across)) {
        flag = FLAG_MARK;
      }
      this.hitFlag = flag;
      this.hitShade = shade;
      return true;
    }

    const tip = bestSign > 0 ? this.finTipRight : this.finTipLeft;
    const [p0, p1, p2] = shape.pectoral;
    if (
      inTriangle(u, a, L * p0, 0, L * p1, 0, L * p2, tip) &&
      // Falcate (sickle) fins get a concave trailing edge.
      !(
        shape.falcate &&
        inTriangle(u, a, L * p1, 0, L * (p1 + (p2 - p1) * 0.33), tip * 0.55, L * p2, tip)
      )
    ) {
      this.hitFlag = this.marked(s, a * bestSign, L, 0) ? FLAG_MARK : FLAG_FIN;
      this.hitShade = 0.38 + 0.22 * (1 - clamp(a / tip, 0, 1));
      return true;
    }
    const [v0, v1, v2, vr] = shape.pelvic;
    if (
      inTriangle(u, a, L * v0, 0, L * v1, 0, L * v2, L * vr) ||
      // Forked caudal fin.
      (shape.caudal > 0 && inTriangle(u, a, L * 0.86, 0, L * 0.95, 0, L, L * shape.caudal))
    ) {
      this.hitFlag = FLAG_FIN;
      this.hitShade = 0.36;
      return true;
    }
    return false;
  }

  /** Whether a body point (s along, v across) falls on a dark marking. */
  private marked(s: number, v: number, L: number, across: number) {
    if (this.shape === SHAPES.tiger) {
      // Wavy vertical bars over the back half, fading out toward the head.
      if (s < 0.2 || s > 0.9 || across > 0.85) return false;
      const wobble = 0.35 * Math.sin((v / L) * 25 + s * 5);
      const bar = Math.sin(s * Math.PI * 2 * 9 + wobble);
      return bar > 0.65 - 0.5 * smoothstep(0.2, 0.42, s) + 0.4 * smoothstep(0.8, 0.9, s);
    }
    if (this.shape === SHAPES.zebra) {
      // Leopard spots on a jittered grid, so they ride along with the skin.
      if (s < 0.07) return false;
      const gu = (s * L) / (L * 0.06);
      const gv = v / (L * 0.045);
      const cu = Math.floor(gu);
      const cv = Math.floor(gv);
      const jx = hash3(cu, cv, this.sharkId) * 0.5 + 0.25;
      const jy = hash3(cu, cv, this.sharkId + 11) * 0.5 + 0.25;
      return (gu - cu - jx) ** 2 + (gv - cv - jy) ** 2 < 0.08;
    }
    return false;
  }

  /**
   * Build a smooth, undulating render spine from the sim spine, then
   * supersample every cell in the shark's bounding box against the shape.
   * Fully covered cells become solid skin; partial cells become edge glyphs.
   */
  private rasterizeSharks(sharks: Shark[], t: number) {
    const {
      sharkFlag,
      sharkShade,
      sharkKey,
      sharkDepth,
      sharkKind,
      sharkShadow,
      spine,
      arc,
      cw,
      ch,
      cols,
      rows,
    } = this;
    sharkFlag.fill(0);
    sharkShadow.fill(0);

    for (const sh of sharks) {
      const L = sh.length;
      const N = SHARK_NODES;
      const n = sh.nodes;
      const swim = clamp(sh.speed / 60, 0.4, 1.4);
      this.shape = SHAPES[sh.kind];
      this.sharkId = sh.id;
      const kindIndex = KINDS.indexOf(sh.kind);

      // Travelling body wave, growing toward the tail.
      for (let k = 0; k < N; k++) {
        const s = k / (N - 1);
        const a = Math.max(0, k - 1);
        const b = Math.min(N - 1, k + 1);
        let tx = n[a * 2] - n[b * 2];
        let ty = n[a * 2 + 1] - n[b * 2 + 1];
        const tl = Math.hypot(tx, ty) || 1;
        tx /= tl;
        ty /= tl;
        const amp = L * 0.05 * (0.1 + 0.9 * Math.pow(s, 1.8)) * swim;
        const off = amp * Math.sin(sh.phase - s * 5.2);
        spine[k * 2] = n[k * 2] - ty * off;
        spine[k * 2 + 1] = n[k * 2 + 1] + tx * off;
      }
      arc[0] = 0;
      for (let k = 1; k < N; k++) {
        arc[k] =
          arc[k - 1] +
          Math.hypot(spine[k * 2] - spine[(k - 1) * 2], spine[k * 2 + 1] - spine[(k - 1) * 2 + 1]);
      }
      this.arcTotal = arc[N - 1] || 1;

      // Bank into turns: the inside fin foreshortens, the outside one spreads.
      const bank = clamp(sh.turn * 0.45, -0.35, 0.35);
      const flex = 1 + 0.06 * Math.sin(sh.phase * 0.5);
      const reachFin = this.shape.pectoral[3];
      this.finTipRight = L * reachFin * flex * (1 - bank);
      this.finTipLeft = L * reachFin * flex * (1 + bank);

      let minX = Infinity;
      let minY = Infinity;
      let maxX = -Infinity;
      let maxY = -Infinity;
      for (let k = 0; k < N; k++) {
        minX = Math.min(minX, spine[k * 2]);
        maxX = Math.max(maxX, spine[k * 2]);
        minY = Math.min(minY, spine[k * 2 + 1]);
        maxY = Math.max(maxY, spine[k * 2 + 1]);
      }
      const reach = L * 0.36;
      const c0 = Math.max(0, Math.floor((minX - reach) / cw));
      const c1 = Math.min(cols - 1, Math.floor((maxX + reach) / cw));
      const r0 = Math.max(0, Math.floor((minY - reach) / ch));
      const r1 = Math.min(rows - 1, Math.floor((maxY + reach) / ch));
      if (c0 > c1 || r0 > r1) continue;

      const skipBeyond = reach + Math.hypot(cw, ch);
      const shadowDc = Math.round((L * 0.035) / cw) + 1;
      const shadowDr = Math.round((L * 0.02) / ch) + 1;

      for (let r = r0; r <= r1; r++) {
        for (let c = c0; c <= c1; c++) {
          const x = c * cw;
          const y = r * ch;
          const centerHit = this.sampleShark(x + cw / 2, y + ch / 2, L);
          if (!centerHit && this.hitDist > skipBeyond) continue;
          const flag = this.hitFlag;
          const shade = this.hitShade;
          const s = this.hitS;
          const v = this.hitV;

          let hits = centerHit ? 1 : 0;
          for (let k = 0; k < SUBSAMPLES.length; k += 2) {
            if (this.sampleShark(x + SUBSAMPLES[k] * cw, y + SUBSAMPLES[k + 1] * ch, L)) hits++;
          }
          if (hits === 0) continue;
          const coverage = hits / 5;

          const i = r * cols + c;
          // Body-anchored key so each glyph rides along with the skin.
          const key =
            (hash3(Math.round(s * 52), Math.round(v / (cw * 0.9)), sh.id) * 4294967295) >>> 0;
          sharkKey[i] = key;
          sharkDepth[i] = sh.depth;
          sharkKind[i] = kindIndex;

          if (centerHit && coverage >= 0.6) {
            sharkFlag[i] = flag;
            const flicker = (hash3(c, r, Math.floor(t * 6)) - 0.5) * 0.06;
            sharkShade[i] = clamp(shade + flicker, 0, 1);
            const sc = c + shadowDc;
            const sr = r + shadowDr;
            if (sc < cols && sr < rows) sharkShadow[sr * cols + sc] = 1;
          } else {
            sharkFlag[i] = FLAG_EDGE;
            sharkShade[i] = coverage;
          }
        }
      }
    }
  }

  private drawSharks(t: number) {
    const { ctx, cols, rows, cw, ch, sharkFlag, sharkShade, sharkKey, sharkDepth, sharkKind } =
      this;
    const n = cols * rows;
    const ink = this.palette.mode === "dark" ? INK.dark : INK.light;
    for (let i = 0; i < n; i++) {
      const flag = sharkFlag[i];
      if (!flag) continue;
      const col = i % cols;
      const row = (i / cols) | 0;
      const x = col * cw;
      const y = row * ch;
      const key = sharkKey[i];
      const shade = sharkShade[i];
      const dim = 0.7 + 0.3 * sharkDepth[i];
      const [tr, tg, tb] = SHAPES[KINDS[sharkKind[i]]].tint;

      // Colours re-roll on a staggered clock so the hide shimmers.
      const tick = Math.floor(t * 2.4 + (key % 997) / 97);
      const roll = hash3(key, tick, 7);

      if (flag === FLAG_EDGE) {
        // Soft, anti-aliased silhouette: no block, just a light glyph.
        const set = shade >= 0.5 ? EDGE_HEAVY : EDGE_LIGHT;
        const g = this.tone((70 + shade * 80) * dim);
        ctx.fillStyle =
          roll < 0.07
            ? ACCENTS[key % ACCENTS.length]
            : `rgb(${Math.round(g * tr)},${Math.round(g * tg)},${Math.min(255, Math.round(g * tb + 6))})`;
        ctx.fillText(set[(key >>> 8) % set.length], x + cw / 2, y + ch / 2 + 1);
        continue;
      }

      let level: number;
      let glyph: string;
      if (flag === FLAG_RIDGE) {
        level = 226;
        glyph = RIDGE_GLYPHS[(key >>> 8) % RIDGE_GLYPHS.length];
      } else if (flag === FLAG_EYE) {
        level = 12;
        glyph = "o";
      } else if (flag === FLAG_MARK) {
        level = 24 + shade * 22;
        glyph = MARK_GLYPHS[(key >>> 8) % MARK_GLYPHS.length];
      } else if (flag === FLAG_FIN) {
        level = 40 + shade * 120;
        glyph = FIN_GLYPHS[(key >>> 8) % FIN_GLYPHS.length];
      } else {
        level = 46 + shade * 160;
        const jitter = ((key >>> 4) % 3) - 1;
        glyph = SKIN_RAMP[clamp(Math.floor((1 - shade) * 8.99) + jitter, 0, SKIN_RAMP.length - 1)];
      }
      // Depth dims toward the background first, then greys invert in light mode.
      level = this.tone(level * dim);
      // Tinted per species; the classic tint is a cool blue-grey, like a back seen through deep water.
      ctx.fillStyle = `rgb(${Math.min(255, Math.round(level * tr))},${Math.min(255, Math.round(level * tg))},${Math.min(255, Math.round(level * tb + 4))})`;
      ctx.fillRect(x + 1, y + 1, cw - 1, ch - 1);

      let color: string;
      if (flag === FLAG_EYE) color = ink.eye;
      else if (flag === FLAG_MARK) {
        const m = this.tone(70);
        color = `rgb(${Math.round(m * tr)},${Math.round(m * tg)},${Math.round(m * tb)})`;
      } else if (roll < 0.09) color = ACCENTS[key % ACCENTS.length];
      else if (roll < 0.13) color = ink.bright;
      else color = shade > 0.55 ? ink.ink : ink.inkSoft;
      ctx.fillStyle = color;
      ctx.fillText(glyph, x + cw / 2, y + ch / 2 + 1);
    }
  }

  // ── fish, dots, heat ────────────────────────────────────────────────

  /**
   * Snap each fish to a cell and bucket it by brightness: far fish are dim,
   * near fish brighter, and turning or panicking fish glint. Nearer fish
   * claim a cell first when two land in the same one.
   */
  private markFish(world: World) {
    const { fishStamp, fishBuckets, sharkFlag } = this;
    const frame = ++this.frameId;
    for (const bucket of fishBuckets) bucket.length = 0;
    for (let pass = 0; pass < 2; pass++) {
      for (const f of world.fish) {
        if (!f.alive || (f.depth > 0.78 ? 0 : 1) !== pass) continue;
        const i = this.cellIndex(f.x, f.y);
        if (i < 0 || fishStamp[i] === frame || sharkFlag[i]) continue;
        fishStamp[i] = frame;
        const glint = clamp(Math.abs(f.turn) / 3.5, 0, 1);
        const v = 0.3 + (0.4 * (f.depth - 0.55)) / 0.45 + 0.3 * Math.max(glint, f.alarm);
        fishBuckets[clamp(Math.floor(v * FISH_ALPHAS.length), 0, FISH_ALPHAS.length - 1)].push(i);
      }
    }
  }

  private drawFish() {
    const { ctx, cw, ch, cols, fishBuckets } = this;
    for (let b = 0; b < fishBuckets.length; b++) {
      const bucket = fishBuckets[b];
      if (!bucket.length) continue;
      ctx.fillStyle = this.fishStyles[b];
      for (const i of bucket) {
        ctx.fillText("-", (i % cols) * cw + cw / 2, ((i / cols) | 0) * ch + ch / 2 + 1);
      }
    }
  }

  private drawHeat() {
    const { ctx, heat, cols, cw, ch } = this;
    for (let i = 0; i < heat.length; i++) {
      const h = heat[i];
      if (h < 0.08) continue;
      const col = i % cols;
      const row = (i / cols) | 0;
      ctx.fillStyle =
        h > 0.5 ? `rgba(255,120,60,${h.toFixed(2)})` : `rgba(220,40,40,${(h * 1.6).toFixed(2)})`;
      ctx.fillText(h > 0.6 ? "*" : "·", col * cw + cw / 2, row * ch + ch / 2 + 1);
    }
  }

  /**
   * Every cell's dot twinkles on its own clock, is modulated by slow
   * caustic light bands, and flares cyan where the wake field is strong.
   * Dots are bucketed by colour/alpha so each bucket is one fillStyle.
   */
  private drawDots(t: number) {
    const {
      ctx,
      cols,
      rows,
      cw,
      ch,
      dotKind,
      dotPhase,
      dotFreq,
      energy,
      sharkFlag,
      sharkShadow,
      fishStamp,
      frameId,
    } = this;
    const { bucketOf, bucketOrder, bucketCounts } = this;
    const n = cols * rows;
    const SKIP = bucketCounts.length - 1;
    bucketCounts.fill(0);

    for (let i = 0; i < n; i++) {
      const e = energy[i];
      const kind = dotKind[i];
      if (sharkFlag[i] || fishStamp[i] === frameId || (kind === 0 && e < 0.45)) {
        bucketOf[i] = SKIP;
        continue;
      }
      let bucket: number;
      if (e > 0.45) {
        const level = clamp(Math.floor((e - 0.45) * 5), 0, ALPHA_LEVELS - 1);
        bucket = WAKE * ALPHA_LEVELS + level;
      } else {
        const col = i % cols;
        const row = (i / cols) | 0;
        // Dots stay bright and crisp; the twinkle is a gentle breathe, not a fade.
        const twinkle = 0.5 + 0.5 * Math.sin(t * dotFreq[i] + dotPhase[i]);
        const caustic =
          0.85 + 0.15 * Math.sin(col * 0.11 + t * 0.35) * Math.sin(row * 0.17 - t * 0.27);
        let v = clamp((0.55 + 0.45 * twinkle) * caustic + e * 2, 0, 1);
        if (sharkShadow[i]) v *= 0.2;
        bucket =
          (kind - 1) * ALPHA_LEVELS + clamp(Math.floor(v * ALPHA_LEVELS), 0, ALPHA_LEVELS - 1);
      }
      bucketOf[i] = bucket;
      bucketCounts[bucket]++;
    }

    // Counting sort cells by bucket.
    const starts = new Int32Array(bucketCounts.length);
    for (let b = 1; b < bucketCounts.length; b++) starts[b] = starts[b - 1] + bucketCounts[b - 1];
    const cursor = starts.slice();
    for (let i = 0; i < n; i++) {
      const b = bucketOf[i];
      if (b !== SKIP) bucketOrder[cursor[b]++] = i;
    }

    const small = Math.max(2, Math.round(cw * 0.19));
    const large = small + 1;
    for (let b = 0; b < SKIP; b++) {
      const count = bucketCounts[b];
      if (!count) continue;
      ctx.fillStyle = this.dotStyles[b];
      const size = b >= WAKE * ALPHA_LEVELS ? large : small;
      const start = starts[b];
      for (let k = start; k < start + count; k++) {
        const i = bucketOrder[k];
        const col = i % cols;
        const row = (i / cols) | 0;
        // Sit the dot low-right in its cell, as in the reference.
        ctx.fillRect(
          Math.round(col * cw + cw * 0.66 - size / 2),
          Math.round(row * ch + ch * 0.64 - size / 2),
          size,
          size,
        );
      }
    }
  }
}
```

```ts
// Ported from https://github.com/stackedlol/ascii-sharks (used with the author's permission).
import { AsciiRenderer, DEFAULT_PALETTE, type OceanPalette } from "./renderer";
import { World, type SharkSpec } from "./world";

const STEP = 1 / 60;
const MAX_STEPS = 4;

/**
 * Wires World + AsciiRenderer to a canvas: fixed-timestep loop, resize,
 * pointer input, pausing off-screen, and prefers-reduced-motion.
 */
export class OceanEngine {
  private world: World;
  private renderer: AsciiRenderer;
  private raf = 0;
  private last = 0;
  private acc = 0;
  private visible = true;
  private reducedMotion: MediaQueryList;
  private colorScheme: MediaQueryList;
  private themeObserver: MutationObserver;
  private resizeObserver: ResizeObserver;
  private intersectionObserver: IntersectionObserver;
  private disposed = false;
  private probe: CanvasRenderingContext2D | null = null;

  constructor(
    private canvas: HTMLCanvasElement,
    sharks?: SharkSpec[],
  ) {
    this.world = new World(sharks);
    this.renderer = new AsciiRenderer(canvas);
    this.renderer.setFontFamily(getComputedStyle(canvas).fontFamily);
    this.renderer.setPalette(this.readPalette());
    this.reducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)");
    this.colorScheme = window.matchMedia("(prefers-color-scheme: dark)");
    this.colorScheme.addEventListener("change", this.onThemeChange);
    this.themeObserver = new MutationObserver(this.onThemeChange);
    this.themeObserver.observe(document.documentElement, {
      attributes: true,
      attributeFilter: ["class", "style", "data-theme"],
    });

    this.resizeObserver = new ResizeObserver(() => this.resize());
    this.resizeObserver.observe(canvas);
    this.intersectionObserver = new IntersectionObserver(([entry]) => {
      this.visible = entry.isIntersecting;
      this.schedule();
    });
    this.intersectionObserver.observe(canvas);

    this.reducedMotion.addEventListener("change", this.onMotionPreference);
    window.addEventListener("pointermove", this.onPointerMove, { passive: true });
    window.addEventListener("pointerdown", this.onPointerDown, { passive: true });
    document.documentElement.addEventListener("pointerleave", this.onPointerLeave);
    window.addEventListener("blur", this.onPointerLeave);

    this.resize();
    // Canvas text can't trigger a webfont load; once the mono face is in, redraw.
    document.fonts?.load(`600 16px ${getComputedStyle(canvas).fontFamily}`).then(() => {
      if (!this.disposed && this.reducedMotion.matches) this.drawStill();
    });
    this.schedule();
  }

  destroy() {
    this.disposed = true;
    cancelAnimationFrame(this.raf);
    this.resizeObserver.disconnect();
    this.intersectionObserver.disconnect();
    this.themeObserver.disconnect();
    this.colorScheme.removeEventListener("change", this.onThemeChange);
    this.reducedMotion.removeEventListener("change", this.onMotionPreference);
    window.removeEventListener("pointermove", this.onPointerMove);
    window.removeEventListener("pointerdown", this.onPointerDown);
    document.documentElement.removeEventListener("pointerleave", this.onPointerLeave);
    window.removeEventListener("blur", this.onPointerLeave);
  }

  private resize() {
    // Layout size, not getBoundingClientRect(): a CSS transform on an ancestor
    // (e.g. a scaled preview) must not shrink the world.
    const width = this.canvas.clientWidth;
    const height = this.canvas.clientHeight;
    if (width < 1 || height < 1) return;
    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    this.renderer.resize(width, height, dpr);
    this.world.resize(width, height, this.renderer.cellWidth, this.renderer.cellHeight);
    if (this.reducedMotion.matches) this.drawStill();
  }

  private schedule() {
    cancelAnimationFrame(this.raf);
    if (this.disposed || !this.visible || this.reducedMotion.matches) return;
    this.last = performance.now();
    this.raf = requestAnimationFrame(this.frame);
  }

  private frame = (now: number) => {
    const dt = Math.min(0.1, (now - this.last) / 1000);
    this.last = now;
    this.acc += dt;
    let steps = 0;
    while (this.acc >= STEP && steps < MAX_STEPS) {
      this.world.step(STEP);
      this.acc -= STEP;
      steps++;
    }
    if (steps === MAX_STEPS) this.acc = 0;
    this.renderer.render(this.world, dt);
    this.raf = requestAnimationFrame(this.frame);
  };

  /** Reduced motion: let the ecosystem settle, then show a single frame. */
  private drawStill() {
    if (this.world.time < 4) {
      for (let i = 0; i < 240; i++) this.world.step(STEP);
    }
    this.renderer.render(this.world, STEP);
  }

  private onMotionPreference = () => {
    if (this.reducedMotion.matches) {
      cancelAnimationFrame(this.raf);
      this.drawStill();
    } else {
      this.schedule();
    }
  };

  /** Background and text colours of the canvas, read from the theme tokens it is styled with. */
  private readPalette(): OceanPalette {
    const style = getComputedStyle(this.canvas);
    const bg = this.toRgba(style.backgroundColor);
    const fg = this.toRgba(style.color);
    // No background colour (transparent): keep the default look.
    if (!bg || !fg || bg[3] === 0) return DEFAULT_PALETTE;
    const luminance = (0.2126 * bg[0] + 0.7152 * bg[1] + 0.0722 * bg[2]) / 255;
    return {
      background: [bg[0], bg[1], bg[2]],
      foreground: [fg[0], fg[1], fg[2]],
      mode: luminance > 0.5 ? "light" : "dark",
    };
  }

  /** Resolves any CSS colour (including oklch()) to rgba bytes via a reusable 1×1 canvas. */
  private toRgba(css: string): Uint8ClampedArray | null {
    if (!this.probe) {
      const canvas = document.createElement("canvas");
      canvas.width = 1;
      canvas.height = 1;
      this.probe = canvas.getContext("2d", { willReadFrequently: true });
    }
    const ctx = this.probe;
    if (!ctx) return null;
    ctx.clearRect(0, 0, 1, 1);
    ctx.fillStyle = "#000";
    ctx.fillStyle = css;
    ctx.fillRect(0, 0, 1, 1);
    return ctx.getImageData(0, 0, 1, 1).data;
  }

  /** Light/dark switch or accent change: re-read the page colours and repaint. */
  private onThemeChange = () => {
    this.renderer.setPalette(this.readPalette());
    if (this.reducedMotion.matches) this.drawStill();
  };

  private toLocal(e: PointerEvent) {
    const rect = this.canvas.getBoundingClientRect();
    // Map screen pixels back to layout pixels when an ancestor is transformed.
    const sx = rect.width > 0 ? this.canvas.clientWidth / rect.width : 1;
    const sy = rect.height > 0 ? this.canvas.clientHeight / rect.height : 1;
    const x = (e.clientX - rect.left) * sx;
    const y = (e.clientY - rect.top) * sy;
    const inside =
      x >= 0 && y >= 0 && x <= this.canvas.clientWidth && y <= this.canvas.clientHeight;
    return { x, y, inside };
  }

  private onPointerMove = (e: PointerEvent) => {
    const { x, y, inside } = this.toLocal(e);
    const p = this.world.pointer;
    p.x = x;
    p.y = y;
    p.active = inside && e.pointerType !== "touch";
  };

  private onPointerDown = (e: PointerEvent) => {
    const { x, y, inside } = this.toLocal(e);
    if (inside && !this.reducedMotion.matches) this.world.splash(x, y);
  };

  private onPointerLeave = () => {
    this.world.pointer.active = false;
  };
}
```

```tsx
"use client";

import { useEffect, useRef } from "react";

import { cn } from "../../lib/utils";
import { OceanEngine } from "./engine";
import type { SharkSpec } from "./world";

export type { SharkKind, SharkSpec } from "./world";

export type AsciiOceanProps = {
  sharks?: SharkSpec[];
  className?: string;
};

/**
 * Glyphs come from the canvas's font-family, so give the canvas (or a parent) a monospace font.
 * Colours come from the canvas's `background-color` and `color`, and follow the site theme live;
 * override them with classes to theme it differently.
 * The roster is fixed for the life of the canvas; change `key` to swap it.
 */
export function AsciiOcean({ sharks, className }: AsciiOceanProps) {
  const canvasRef = useRef<HTMLCanvasElement>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const engine = new OceanEngine(canvas, sharks);
    return () => engine.destroy();
    // The roster is read once per canvas; a new array identity must not restart the simulation.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  return (
    <canvas
      ref={canvasRef}
      aria-hidden="true"
      className={cn("bg-background text-foreground block size-full font-mono", className)}
    />
  );
}
```
